1"""Resolves scenario names to classes for every caller. 

2 

3Two sources feed it: the registry of scenarios shipped here, and scenarios 

4declared by other installed distributions under the versioned 

5``glossogen.scenarios.v<N>`` entry-point group. Callers ask by name and do not 

6know which source answered. 

7 

8Scenarios shipped here are imported when the registry module is. An external 

9scenario's class is imported the first time someone asks for it, so listing the 

10available names stays cheap. 

11""" 

12

30# Problems already reported this process, so a per-request listing does not repeat 

31# them. Keyed by the problem and the declaration it concerns, not by name alone, 

32# so a plug-in that changes its declaration is reported again. 

33_ALREADY_WARNED: set[tuple[str, ...]] = set()

36def available_scenario_names() -> list[str]: 

37 """Return every scenario name that can be run, built-in and external. 

38 

39 Imports no scenario class: external names come from installed metadata. 

40 """ 

41 declarations = scenario_declarations() 

42 _report_unusable(declarations=declarations) 

43 return sorted(set(SCENARIO_REGISTRY) | set(declarations.current))

46def find_scenario_class(name: str) -> type[SimulationScenario] | None: 

47 """Return the scenario class registered under ``name``, or None if there is none. 

48 

49 The soft half of the pair, for callers that treat not having a scenario as an 

50 ordinary outcome: a run whose recorded scenario is no longer installed, or a 

51 run-detail page for one that a plug-in has since broken. Those callers degrade, 

52 reporting no primary channels or a 404, and a raise from here would turn each 

53 into a 500 on the one path the design otherwise keeps serving. 

54 

55 So a misdeclared plug-in is logged and answered with None, the same tolerance 

56 :func:`iter_scenario_classes` applies for the same reason. Use 

57 :func:`get_scenario_class` where the caller asked for this scenario by name and 

58 wants to be told why it cannot have it. 

59 """ 

60 built_in = SCENARIO_REGISTRY.get(name) 

61 if built_in is not None: 

62 return built_in 

63 entry_point = scenario_entry_points().get(name) 

64 if entry_point is None: 

65 return None 

66 try: 

67 return _load_external(name=name, entry_point=entry_point) 

68 except ValueError: 

69 logger.exception("Scenario %r is installed but could not be loaded", name) 

70 return None

73def get_scenario_class(name: str) -> type[SimulationScenario]: 

74 """Return the scenario class registered under ``name``. 

75 

76 Raises ValueError if the name does not match any registered scenario, or names 

77 one that is installed and cannot be loaded. Resolves without going through 

78 :func:`find_scenario_class`, which answers None in that second case and would 

79 lose the explanation. 

80 """ 

81 built_in = SCENARIO_REGISTRY.get(name) 

82 if built_in is not None: 

83 return built_in 

84 entry_point = scenario_entry_points().get(name) 

85 if entry_point is not None: 

86 return _load_external(name=name, entry_point=entry_point) 

87 other_group = scenarios_declared_under_other_groups().get(name) 

88 if other_group is not None: 

89 raise ValueError( 

90 f"Scenario '{name}' is installed, but declared under the entry-point group " 

91 f"'{other_group}', which this glossogen does not read. It reads " 

92 f"'{SCENARIO_ENTRY_POINT_GROUP}' (scenario contract version " 

93 f"{SCENARIO_API_VERSION}). Upgrade whichever side is older, or correct the " 

94 "group name." 

95 ) 

96 available = ", ".join(available_scenario_names()) 

97 raise ValueError(f"Unknown scenario: '{name}'. Available scenarios: {available}")

100def iter_scenario_classes() -> list[tuple[str, type[SimulationScenario]]]: 

101 """Return every loadable scenario as ``(name, class)``, ordered by name. 

102 

103 Unlike :func:`available_scenario_names` this imports every external 

104 scenario, so use it only where the classes themselves are needed, such as 

105 listing each scenario's metrics and presets. Installed metadata is read once 

106 for the whole listing rather than once per name. 

107 

108 An external scenario that cannot be loaded is logged and left out, rather 

109 than raising. This backs the scenario list, so one unusable third-party 

110 plug-in would otherwise take the list down for every scenario shipped here 

111 and leave the caller no way to run any of them. Asking for that scenario by 

112 name still raises, because then there is nothing else the caller wanted. 

113 """ 

114 declarations = scenario_declarations() 

115 _report_unusable(declarations=declarations) 

116 external = declarations.current 

117 resolved: list[tuple[str, type[SimulationScenario]]] = list(SCENARIO_REGISTRY.items()) 

118 for name in set(external) - set(SCENARIO_REGISTRY): 

119 try: 

120 resolved.append((name, _load_external(name=name, entry_point=external[name]))) 

121 except ValueError: 

122 logger.exception("Leaving scenario %r out of the listing", name) 

123 return sorted(resolved, key=lambda pair: pair[0])

126def _report_unusable(declarations: ScenarioDeclarations) -> None: 

127 """Report every installed scenario declaration this platform will not use. 

128 

129 Both cases are silent by nature: a shadowed name resolves to somebody else's 

130 scenario, and one declared under a group nothing reads is missing from the list 

131 with nothing said. An operator looks at the listing first after installing a 

132 plug-in, so both are reported there. 

133 

134 Each distinct problem is reported once per process. Listing backs a web endpoint 

135 and an MCP tool, and repeating the same warning per request would bury it in its 

136 own copies. 

137 """ 

138 for name, entry_point in declarations.current.items(): 

139 if name not in SCENARIO_REGISTRY: 

140 continue 

141 _warn_once( 

142 key=("shadowed", name, entry_point.value), 

143 message=( 

144 "Scenario %r is already provided by glossogen; ignoring the one declared by %s" 

145 ), 

146 args=(name, entry_point.value), 

147 ) 

148 for name, group in declarations.other_groups.items(): 

149 _warn_once( 

150 key=("other-group", name, group), 

151 message=( 

152 "Scenario %r is declared under entry-point group %r, which this glossogen " 

153 "does not read; it reads %r. Not listing it." 

154 ), 

155 args=(name, group, SCENARIO_ENTRY_POINT_GROUP), 

156 )

159def forget_reported_problems() -> None: 

160 """Forget which problems have been reported, so they are reported again. 

161 

162 The suppression below is per process, so a warning fires only the first time its 

163 key is seen. In a running server that is the intent. In a test suite it couples 

164 one test's assertions to another's choice of scenario name, so tests call this 

165 between cases. 

166 """ 

167 _ALREADY_WARNED.clear()

170def _warn_once(key: tuple[str, ...], message: str, args: tuple[str, ...]) -> None: 

171 """Log a warning the first time this exact problem is seen in this process.""" 

172 if key in _ALREADY_WARNED: 

173 return 

174 _ALREADY_WARNED.add(key) 

175 logger.warning(message, *args)

178def _load_external(name: str, entry_point: EntryPoint) -> type[SimulationScenario]: 

179 """Import an externally-declared scenario class and check it over. 

180 

181 Every failure here is the plug-in's problem rather than the platform's, so 

182 each one raises ValueError naming the entry point. Unlike a metric, which is 

183 skipped so the others still run, a scenario that cannot be loaded is one the 

184 caller asked for by name, so there is nothing to fall back to. 

185 """ 

186 try: 

187 loaded = entry_point.load() 

188 except Exception as exc: 

189 raise ValueError( 

190 f"Scenario entry point '{name}' ({entry_point.value}) failed to import: {exc}" 

191 ) from exc 

192 if not isinstance(loaded, type) or not issubclass(loaded, SimulationScenario): 

193 raise ValueError( 

194 f"Scenario entry point '{name}' ({entry_point.value}) " 

195 "does not name a SimulationScenario subclass" 

196 ) 

197 check_entry_point_declaration(name=name, entry_point=entry_point, loaded=loaded) 

198 return loaded

218def _check_defined_in_a_submodule( 

219 name: str, 

220 entry_point: EntryPoint, 

221 loaded: type[SimulationScenario], 

222) -> None: 

223 """Refuse an entry point that points at a package rather than a module. 

224 

225 Event discovery reads the package a scenario lives in from the entry-point 

226 string, without importing anything, because it runs while 

227 ``glossogen.models.event`` is mid-import. That read assumes the string names a 

228 module inside the package, and takes everything before the last dot. Point it 

229 at the package itself and the package is misread as its own parent, the 

230 ``events`` module looks absent, and the scenario's event types never reach the 

231 parser: the run writes fine and its JSONL will not parse back afterwards. 

232 

233 Checked against the module the entry point names, not the module the class is 

234 defined in. Those differ whenever the package's ``__init__`` re-exports the 

235 class, which is ordinary Python packaging: the class itself then looks fine 

236 while discovery still reads from the wrong place. 

237 """ 

238 module = sys.modules.get(entry_point.module) 

239 if module is None or not hasattr(module, "__path__"): 

240 return 

241 if loaded.__module__ != entry_point.module: 

242 # The class is in a submodule and the package re-exports it, so the entry 

243 # point only has to name where it was defined. 

244 remedy = ( 

245 f"Point it at the module defining the class ('{loaded.__module__}:{loaded.__name__}')" 

246 ) 

247 else: 

248 remedy = ( 

249 "Move the class into a submodule and point the entry point at it " 

250 f"('{entry_point.module}.scenario:{loaded.__name__}')" 

251 ) 

252 raise ValueError( 

253 f"Scenario entry point '{name}' ({entry_point.value}) points at the " 

254 f"'{entry_point.module}' package rather than a module inside it, which glossogen " 

255 f"cannot discover events for. {remedy}, and keep the package's __init__ empty." 

256 )

259def _check_reported_name( 

260 name: str, 

261 entry_point: EntryPoint, 

262 loaded: type[SimulationScenario], 

263) -> None: 

264 """Refuse a class that answers to a different name than it is registered under. 

265 

266 ``name()`` is what a run directory is named after and what 

267 ``SimulationStarted`` records, while the name here is what ``glossogen run``, 

268 ``evaluate`` and the resume flows look a run up by. Let the two disagree and 

269 a run launched as ``reactor_purge`` lands in ``runs/purge/``, where none of 

270 those commands will find it again. Nothing about the run looks wrong at the 

271 time. 

272 """ 

273 try: 

274 reported = loaded.name() 

275 except Exception as exc: 

276 raise ValueError( 

277 f"Scenario entry point '{name}' ({entry_point.value}) " 

278 f"could not report its own name: {exc}" 

279 ) from exc 

280 if reported != name: 

281 raise ValueError( 

282 f"Scenario entry point '{name}' names a class that calls itself {reported!r}. " 

283 "The entry-point name and the scenario's name() must match, because run " 

284 "directories are named after name() and every later command looks a run up " 

285 "by the name it was launched with. Rename the entry point to " 

286 f"{reported!r}, or override name() to return {name!r}." 

287 )
