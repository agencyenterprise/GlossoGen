1"""Read the `.env` beside the project a command was run from. 

2 

3``python-dotenv``'s ``load_dotenv()`` locates the file by walking up from the 

4module that called it, not from the working directory. Inside a checkout those 

5are the same place, so the repo's own `.env` is found either way. Installed as a 

6dependency they are not: the walk starts in ``site-packages`` and finds nothing, 

7so a project's `.env` is silently ignored and its `ANTHROPIC_API_KEY` with it. 

8 

9Walking up from the working directory instead covers both, and keeps a command 

10run from a subdirectory reading the project's file. 

11""" 

12 

13import logging 

14from pathlib import Path 

15 

16from dotenv import find_dotenv, load_dotenv 

17 

18logger = logging.getLogger(__name__) 

19 

20 

21def load_env_from_working_directory() -> Path | None: 

22 """Load the nearest `.env` at or above the working directory. 

23 

24 Returns the file that was read, or None when there is none. Values already 

25 in the environment win, which is what lets a variable set on the command 

26 line override the file for one run. 

27 """ 

28 located = find_dotenv(usecwd=True) 

29 if not located: 

30 return None 

31 load_dotenv(dotenv_path=located) 

32 logger.debug("Loaded environment from %s", located) 

33 return Path(located)
