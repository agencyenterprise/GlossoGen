1"""The event types one scenario's ``events`` module defines, and a parser for its runs. 

2 

3The module-level parser in ``glossogen.models.event`` was built while that module 

4was importing, so it cannot see a scenario loaded from a path afterwards. A run 

5logs only the platform's events and its own scenario's, so a parser over exactly 

6those reads any run, whichever way its scenario was loaded.

19def events_module_name(scenario_cls: type[SimulationScenario]) -> str | None: 

20 """Return the dotted name of the scenario's ``events`` module, or None. 

21 

22 Asked of the scenario module's own import spec. ``parent`` is the value Python 

23 resolves a relative import against, so it answers for a class in a 

24 ``scenario`` submodule and for one defined in the package's ``__init__``. 

25 None when the class is in a top-level module belonging to no package. 

26 """ 

27 spec = importlib.import_module(scenario_cls.__module__).__spec__ 

28 if spec is None or not spec.parent: 28 ↛ 29line 28 didn't jump to line 29 because the condition on line 28 was never true

29 return None 

30 return f"{spec.parent}.events"

33def scenario_event_types(scenario_cls: type[SimulationScenario]) -> tuple[type[EventBase], ...]: 

34 """Return the event types the scenario's ``events`` module defines, importing it. 

35 

36 A scenario with no ``events`` module defines none. 

37 """ 

38 name = events_module_name(scenario_cls=scenario_cls) 

39 if name is None or importlib.util.find_spec(name) is None: 

40 return () 

41 module = importlib.import_module(name) 

42 return tuple( 

43 attribute 

44 for attribute in vars(module).values() 

45 if isinstance(attribute, type) 

46 and issubclass(attribute, EventBase) 

47 and attribute is not EventBase 

48 and attribute.__module__ == module.__name__ 

49 )
