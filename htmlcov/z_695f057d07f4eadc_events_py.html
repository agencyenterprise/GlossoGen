1"""Pydantic event types specific to the spot_the_difference scenario. 

2 

3The event log serializes the per-round ground truth (both scenes plus the 

4planted differences) and the LLM judge's verdict on every ``submit_differences`` 

5call (which planted differences the team's submission identified, how many of 

6its items matched nothing, and the team's character count at submission). 

7""" 

8 

9from typing import Literal 

10 

11from pydantic import BaseModel 

12 

13from glossogen.models.event_base import EventBase 

14 

15 

16class SpotObject(BaseModel): 

17 """One scene object: a shape/color/size bundle at a grid cell. 

18 

19 ``column`` / ``row`` are the internal geometry (never shown to agents); 

20 ``region`` is the coarse 3x3 area the agents actually see. 

21 """ 

22 

23 shape: str 

24 color: str 

25 size: str 

26 column: int 

27 row: int 

28 region: str 

29 

30 

31class SpotPlantedDifference(BaseModel): 

32 """One ground-truth difference between scene A and scene B. 

33 

34 ``scene_a_object`` / ``scene_b_object`` is ``None`` for an added object 

35 (absent from A) or a removed object (absent from B). ``attribute_name`` 

36 names the changed dimension for ``attribute_changed`` and is ``None`` 

37 otherwise. 

38 """ 

39 

40 kind: str 

41 description: str 

42 scene_a_object: SpotObject | None 

43 scene_b_object: SpotObject | None 

44 attribute_name: str | None 

45 

46 

47class SpotTheDifferenceCaseStarted(EventBase): 

48 """Emitted once at round start with the full ground-truth scene pair. 

49 

50 The left viewer sees ``scene_a`` and the right viewer sees ``scene_b``; 

51 neither sees the other scene nor ``differences``. 

52 """ 

53 

54 event_type: Literal["spot_the_difference_case_started"] = "spot_the_difference_case_started" 

55 case_number: int 

56 grid_size: int 

57 round_time_budget_seconds: int 

58 difference_count: int 

59 scene_a: list[SpotObject] 

60 scene_b: list[SpotObject] 

61 differences: list[SpotPlantedDifference] 

62 

63 

64class DifferenceSubmissionJudged(EventBase): 

65 """Emitted after the LLM judge rules on a ``submit_differences`` call. 

66 

67 ``matched_difference_indices`` are the 1-based indices into the round's 

68 planted differences that the team's submission correctly identified; 

69 ``false_positive_count`` is how many submitted items matched no planted 

70 difference. ``found_all`` is the correctness gate (every planted 

71 difference identified and no false positives). 

72 """ 

73 

74 event_type: Literal["difference_submission_judged"] = "difference_submission_judged" 

75 agent_id: str 

76 team_id: str 

77 submitted_items: list[str] 

78 matched_difference_indices: list[int] 

79 false_positive_count: int 

80 found_all: bool 

81 characters_at_submission: int 

82 judge_explanation: str
