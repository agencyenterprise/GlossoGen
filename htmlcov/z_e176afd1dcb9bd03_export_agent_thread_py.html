1"""Orchestrate exporting one agent's thread from a finished run into a provider request. 

2 

3Reuses the protocol probe's reconstruction path: ``build_message_history`` with the 

4same exclusive ``cutoff_round`` semantics and ``build_full_system_prompt``, then 

5serializes the result with ``provider_thread_serializer`` into a ``ThreadExport``. 

6``export_agent_thread_from_run_dir`` loads the JSONL itself; ``export_agent_thread`` 

7takes already-loaded events for callers (the web server) that have them in hand. 

8"""

29ThreadExportFormat = Literal["anthropic_messages", "openai_chat"]

32def default_format_for_provider(provider: str) -> ThreadExportFormat: 

33 """Pick the export format matching the provider the agent ran under. 

34 

35 Anthropic agents serialize to the Anthropic Messages shape; every other 

36 provider (OpenAI and the OpenAI-compatible self-hosted backend) serializes 

37 to the OpenAI Chat shape. 

38 """ 

39 if provider == "anthropic": 

40 return "anthropic_messages" 

41 return "openai_chat"

44def _resolve_agent_config(agent_configs: list[AgentConfig], agent_id: str) -> AgentConfig: 

45 """Return the latest ``AgentConfig`` registered for ``agent_id``. 

46 

47 Cross-run / in-run-swap runs register the same ``agent_id`` more than once; 

48 the last registration carries the model and system prompt the agent ended 

49 the run under, matching the protocol probe's dedup choice. 

50 """ 

51 resolved: AgentConfig | None = None 

52 for config in agent_configs: 

53 if config.agent_id == agent_id: 

54 resolved = config 

55 if resolved is None: 

56 available = ", ".join(sorted({config.agent_id for config in agent_configs})) or "<none>" 

57 raise ValueError(f"No agent {agent_id!r} in run (available agent_ids: {available})") 

58 return resolved

61def _rounds_covered_label(cutoff_round: int | None, events: list[SimulationEvent]) -> str: 

62 """Human-readable description of which rounds the exported history spans.""" 

63 if cutoff_round is None: 

64 last_round = max((event.round_number for event in events), default=0) 

65 return f"1-{last_round}" 

66 if cutoff_round <= 1: 

67 return "none" 

68 return f"1-{cutoff_round - 1}"

71def export_agent_thread( 

72 events: list[SimulationEvent], 

73 agent_configs: list[AgentConfig], 

74 run_id: str, 

75 agent_id: str, 

76 cutoff_round: int | None, 

77 output_format: ThreadExportFormat, 

78 include_thinking: bool, 

79 flatten_tools: bool, 

80) -> ThreadExport: 

81 """Reconstruct ``agent_id``'s thread and serialize it into a provider request body. 

82 

83 ``cutoff_round`` is the same exclusive cutoff the protocol probe uses: ``R`` 

84 keeps rounds ``1..R-1`` (pass ``R+1`` to capture the end of round R); 

85 ``None`` exports the full end-of-run thread. 

86 """ 

87 agent_config = _resolve_agent_config(agent_configs=agent_configs, agent_id=agent_id) 

88 full_system_prompt = build_full_system_prompt( 

89 base_prompt=agent_config.system_prompt, 

90 role_name=agent_config.role_name, 

91 ) 

92 history = build_message_history( 

93 events=events, 

94 agent_id=agent_id, 

95 system_prompt=full_system_prompt, 

96 target_timestamp=resolve_history_timestamp(events=events), 

97 cutoff_round=cutoff_round, 

98 tool_calls_only=False, 

99 channel_visibility={}, 

100 filter_below_round=None, 

101 split_parallel_tool_calls=False, 

102 ) 

103 if not history: 

104 raise ValueError( 

105 f"Reconstructed history for agent {agent_id!r} is empty " 

106 f"(cutoff_round={cutoff_round}); nothing to export." 

107 ) 

108 

109 if output_format == "anthropic_messages": 

110 request = to_anthropic_request( 

111 messages=history, 

112 model=agent_config.model, 

113 include_thinking=include_thinking, 

114 flatten_tools=flatten_tools, 

115 ) 

116 else: 

117 request = to_openai_request( 

118 messages=history, 

119 model=agent_config.model, 

120 include_thinking=include_thinking, 

121 flatten_tools=flatten_tools, 

122 ) 

123 

124 meta = ThreadExportMeta( 

125 run_id=run_id, 

126 agent_id=agent_id, 

127 role_name=agent_config.role_name, 

128 model=agent_config.model, 

129 provider=agent_config.provider, 

130 cutoff_round=cutoff_round, 

131 rounds_covered=_rounds_covered_label(cutoff_round=cutoff_round, events=events), 

132 num_messages=len(request.messages), 

133 format=output_format, 

134 thinking_included=include_thinking, 

135 tools_flattened=flatten_tools, 

136 exported_at=datetime.now(tz=timezone.utc), 

137 ) 

138 logger.info( 

139 "Exported thread for agent %s: format=%s cutoff_round=%s messages=%d " 

140 "(self_hosted=%s, thinking=%s, flatten_tools=%s)", 

141 agent_id, 

142 output_format, 

143 cutoff_round, 

144 len(request.messages), 

145 agent_config.provider == SELF_HOSTED_PROVIDER, 

146 include_thinking, 

147 flatten_tools, 

148 ) 

149 return ThreadExport(meta=meta, request=request)

152async def export_agent_thread_from_run_dir( 

153 run_dir: Path, 

154 scenario_name: str, 

155 agent_id: str, 

156 cutoff_round: int | None, 

157 output_format: ThreadExportFormat | None, 

158 include_thinking: bool, 

159 flatten_tools: bool, 

160) -> ThreadExport: 

161 """Load a run's JSONL and export one agent's thread as a provider request. 

162 

163 ``output_format=None`` defaults to the format matching the agent's own 

164 provider. ``run_id`` is derived as ``<scenario>/<run_dir_name>``. 

165 """ 

166 log_path = run_dir / f"{scenario_name}.jsonl" 

167 events = await load_events(log_path=log_path) 

168 agent_configs = extract_agent_configs(events=events) 

169 agent_config = _resolve_agent_config(agent_configs=agent_configs, agent_id=agent_id) 

170 if output_format is None: 

171 resolved_format = default_format_for_provider(provider=agent_config.provider) 

172 else: 

173 resolved_format = output_format 

174 return export_agent_thread( 

175 events=events, 

176 agent_configs=agent_configs, 

177 run_id=f"{run_dir.parent.name}/{run_dir.name}", 

178 agent_id=agent_id, 

179 cutoff_round=cutoff_round, 

180 output_format=resolved_format, 

181 include_thinking=include_thinking, 

182 flatten_tools=flatten_tools, 

183 )
