1"""Async event logger that writes simulation events as newline-delimited JSON to a file. 

2 

3Publishes each logged event to the ``EventBus`` it is constructed with, for 

4real-time delivery to SSE subscribers. Logs only the event classes it is built 

5with, the platform's and the running scenario's declared ones, so every line it 

6writes is one a reader of the run can parse. An ``UnrecognizedEvent``, which a 

7reader builds from a line it could not parse, is refused too: a derived run keeps 

8its source's lines through ``rewrite_run_jsonl`` and never logs one through here. 

9"""

29class EventLogger: 

30 """Writes ``SimulationEvent`` objects as newline-delimited JSON (JSONL) to a binary file. 

31 

32 ``allowed_event_types`` are the classes it logs: the platform's events and the 

33 ones the running scenario declares in ``event_types()`` 

34 (:func:`glossogen.scenario_contract.event_types.loggable_event_types`). A run 

35 is read back with exactly those, so any other class is refused.

71 async def log(self, event: SimulationEvent) -> None: 

72 """Serialize ``event`` to JSON, write it as a single line, and flush. 

73 

74 Also publishes the event to the event bus for live streaming. 

75 Raises :class:`UndeclaredEventType` for an event class the logger does not 

76 accept, and ``RuntimeError`` if the logger has not been opened. 

77 """ 

78 if isinstance(event, UnrecognizedEvent): 

79 raise UndeclaredEventType( 

80 f"Refusing to log an UnrecognizedEvent (event_type {event.event_type!r}): " 

81 "it is a line read back from a run's log, not an event of this run. A " 

82 "derived run keeps its source's lines through rewrite_run_jsonl instead." 

83 ) 

84 event_cls = type(event) 

85 if event_cls not in self._allowed_event_types: 

86 raise UndeclaredEventType( 

87 f"Refusing to log {event_cls.__module__}.{event_cls.__qualname__} " 

88 f"(event_type {fixed_event_type(event_cls=event_cls)!r}): it is neither " 

89 "a platform event nor one the running scenario returns from " 

90 "event_types(), so a reader of this run would read it as an " 

91 "UnrecognizedEvent. Add it to the scenario's event_types()." 

92 ) 

93 if self._file is None: 

94 raise RuntimeError("EventLogger is not open. Call open() first.") 

95 event_dict = event.model_dump(mode="json") 

96 data = orjson.dumps(event_dict) + b"\n" 

97 async with self._write_lock: 

98 await self._file.write(data) 

99 await self._file.flush() 

100 self._event_bus.publish(event=event_dict)
