1"""Single-shot structured probe of an agent's pydantic-ai instance against a reconstructed history. 

2 

3Given an agent's original ``model``/``provider``, a reconstructed message 

4history, and a probe prompt, ``run_structured_probe`` builds a tool-less 

5pydantic-ai ``Agent`` with the caller's ``output_type`` and runs one 

6``agent.run(...)`` call. No MCP server, no game clock, no subprocess, just one 

7LLM round-trip. It returns the validated structured output together with the 

8token usage of the call so the caller can aggregate cost per ``(model, 

9provider)``. 

10 

11``run_protocol_probe`` is the convenience wrapper used by the protocol_probe 

12metric: it assembles the ``[_PROBE_INTRO, CachePoint(), probe_prompt]`` user 

13prompt (so providers that support prompt caching place the cache breakpoint 

14between the constant intro and the varying probe text) and asks for a 

15``ProtocolProbeOutput``. Other metrics call ``run_structured_probe`` directly 

16with their own ``output_type``. 

17"""

40# Anthropic reasoning models (e.g. claude-opus-4-7) spend output tokens on internal 

41# thinking; without an explicit cap the small Anthropic default is exhausted before 

42# the structured-output tool call is emitted, yielding an empty ``{}`` and a missing- 

43# field validation error. The OpenAI branch of ``default_pydantic_ai_settings`` already 

44# sets a generous cap for the same reason; mirror it for the probe's Anthropic calls.

68async def run_structured_probe( 

69 agent_id: str, 

70 role_name: str, 

71 full_system_prompt: str, 

72 model: str, 

73 provider: str, 

74 message_history: list[ModelMessage], 

75 user_prompt_parts: Sequence[str | CachePoint], 

76 output_type: type[ProbeOutputT], 

77) -> StructuredProbeResult[ProbeOutputT]: 

78 """Run one structured probe call against the agent and return output + usage. 

79 

80 Builds a fresh ``Agent`` with no tools and the caller's ``output_type`` so 

81 the LLM produces a validated structured response. The caller passes the 

82 agent's full system prompt: the ``base_prompt + communication-protocol 

83 suffix`` composition the runner applied at simulation time, so 

84 reconstruction and the new ``Agent`` construction stay consistent. 

85 """ 

86 agent: Agent[None, ProbeOutputT] = Agent( 

87 model=build_pydantic_ai_model(model=model, provider=provider), 

88 system_prompt=full_system_prompt, 

89 output_type=output_type, 

90 model_settings=_probe_model_settings(provider=provider), 

91 ) 

92 logger.info( 

93 "Probing agent %s (%s) under model=%s provider=%s history_len=%d output=%s", 

94 agent_id, 

95 role_name, 

96 model, 

97 provider, 

98 len(message_history), 

99 output_type.__name__, 

100 ) 

101 result = await agent.run( 

102 user_prompt=list(user_prompt_parts), 

103 message_history=message_history, 

104 usage_limits=UsageLimits(request_limit=None), 

105 ) 

106 return StructuredProbeResult( 

107 output=result.output, 

108 usage=_run_usage_to_evaluation_token_usage(run_usage=result.usage), 

109 )

112async def run_protocol_probe( 

113 agent_id: str, 

114 role_name: str, 

115 full_system_prompt: str, 

116 model: str, 

117 provider: str, 

118 message_history: list[ModelMessage],

120) -> StructuredProbeResult[ProtocolProbeOutput]: 

121 """Probe the agent for the ``#link`` message it would send for a hypothetical input. 

122 

123 Assembles ``[_PROBE_INTRO, CachePoint(), probe_prompt]`` so replicas sharing 

124 the same ``system + history + _PROBE_INTRO`` prefix hit the prompt cache, and 

125 asks for a ``ProtocolProbeOutput`` (reasoning + message). 

126 """ 

127 return await run_structured_probe( 

128 agent_id=agent_id, 

129 role_name=role_name, 

130 full_system_prompt=full_system_prompt, 

131 model=model, 

132 provider=provider, 

133 message_history=message_history, 

134 user_prompt_parts=[_PROBE_INTRO, CachePoint(), probe_prompt], 

135 output_type=ProtocolProbeOutput, 

136 )

139def _run_usage_to_evaluation_token_usage(run_usage: RunUsage) -> EvaluationTokenUsage: 

140 """Project pydantic-ai's ``RunUsage`` onto the evaluation cost token model. 

141 

142 ``RunUsage.input_tokens`` is the *total* prompt token count (non-cached + 

143 cache_read + cache_creation), which differs from the Anthropic API's 

144 ``input_tokens`` (after-breakpoint only). ``compute_evaluation_cost`` 

145 expects ``EvaluationTokenUsage.input_tokens`` to be the non-cached 

146 portion only, so we subtract here to match. 

147 """ 

148 non_cached_input = max( 

149 0, 

150 run_usage.input_tokens - run_usage.cache_read_tokens - run_usage.cache_write_tokens, 

151 ) 

152 return EvaluationTokenUsage( 

153 input_tokens=non_cached_input, 

154 output_tokens=run_usage.output_tokens, 

155 cache_read_input_tokens=run_usage.cache_read_tokens, 

156 cache_creation_input_tokens=run_usage.cache_write_tokens, 

157 )
