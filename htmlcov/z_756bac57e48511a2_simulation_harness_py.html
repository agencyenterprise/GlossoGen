58@dataclass(frozen=True) 

59class SimulationResult: 

60 """A finished run: the events it wrote, where, and the scenario that ran. 

61 

62 The scenario is kept because some of what a run decides never reaches the 

63 event log. Whether the discussion phase was left open, for one, is world 

64 state that shuts the task channel without recording anything. 

65 """ 

66 

67 events: list[dict[str, Any]] 

68 log_path: Path 

69 scenario: SimulationScenario

71 def of_type(self, *, event_type: str) -> list[dict[str, Any]]: 

72 """Return every event of one type, in the order logged.""" 

73 return [e for e in self.events if e.get("event_type") == event_type] 

74 

75 def types(self) -> list[str]: 

76 """Return the event types present, deduplicated, in first-seen order.""" 

77 seen: list[str] = [] 

78 for event in self.events: 

79 kind = str(event.get("event_type")) 

80 if kind not in seen: 

81 seen.append(kind) 

82 return seen

84 def tool_calls(self, *, tool_name: str) -> list[dict[str, Any]]: 

85 """Return every invocation of one tool.""" 

86 return [ 

87 e 

88 for e in self.of_type(event_type="tool_call_invoked") 

89 if e.get("tool_name") == tool_name 

90 ]

92 def failed_tool_calls(self) -> list[tuple[str, str]]: 

93 """Return (tool_name, result) for every tool call that reported an error. 

94 

95 Tool failures are returned to the agent as text rather than raised, so a 

96 run completes happily with every call broken. Asserting invocation alone 

97 would not notice. 

98 """ 

99 failures: list[tuple[str, str]] = [] 

100 for event in self.of_type(event_type="tool_result_received"): 

101 result = str(event.get("result", "")) 

102 if result.startswith("Error executing tool") or "validation error" in result: 102 ↛ 103line 102 didn't jump to line 103 because the condition on line 102 was never true

103 failures.append((str(event.get("tool_name")), result[:160])) 

104 return failures

106 def conflicted_sends(self) -> list[str]: 

107 """Return the results of sends rejected by optimistic concurrency. 

108 

109 A conflicted send is not an error: the tool returns ``status="conflict"`` 

110 and the message is never delivered. Nothing raises, so a test asserting 

111 only that ``send_message`` was invoked would not notice the message 

112 going nowhere. 

113 """ 

114 out: list[str] = [] 

115 for event in self.of_type(event_type="tool_result_received"): 

116 if event.get("tool_name") != "send_message": 

117 continue 

118 result = str(event.get("result", "")) 

119 if '"conflict"' in result or "conflict" in result[:60]: 119 ↛ 120line 119 didn't jump to line 120 because the condition on line 119 was never true

120 out.append(result[:160]) 

121 return out

123 def messages_on(self, *, channel_id: str) -> list[dict[str, Any]]: 

124 """Return the messages sent to one channel.""" 

125 out: list[dict[str, Any]] = [] 

126 for event in self.of_type(event_type="message_sent"): 

127 message: dict[str, Any] | None = event.get("message") 

128 if message is not None and message.get("channel_id") == channel_id: 

129 out.append(message) 

130 return out

133def never_times_out(phase_age: float, limit: float) -> bool: 

134 """A phase is never over on time alone: it ends when the agents finish. 

135 

136 Scripted agents take as long as the machine takes, so a limit that fires 

137 first truncates the run and changes what the scenario decided. Saying the 

138 limit never fires is what makes the outcome the same everywhere. 

139 """ 

140 _ = phase_age, limit 

141 return False

146 

147 The agents in such a run never park, so nothing else would end a phase. 

148 """ 

149 _ = phase_age, limit

153async def run_simulation( 

154 *, 

155 scenario: SimulationScenario, 

156 scripts: Mapping[str, Sequence[ScriptedTurn]], 

157 tmp_path: Path, 

158 monkeypatch: pytest.MonkeyPatch, 

159 phase_timed_out: PhaseTimeoutCheck, 

160) -> SimulationResult: 

161 """Run ``scenario`` to completion with each agent following its script. 

162 

163 ``scripts`` maps agent_id to the turns that agent takes, one per cycle. An 

164 agent that runs out of script raises, so a test cannot quietly pass while an 

165 agent loops somewhere nobody described. 

166 

167 ``phase_timed_out`` decides when a phase is over on time alone. Pass 

168 ``never_times_out`` for a run whose phases should end because the agents 

169 finished, and ``always_timed_out`` for one that has to exercise the 

170 timeout. Neither waits, which is what keeps the answer the same on a loaded 

171 machine as on an idle one. 

172 """ 

173 log_path = tmp_path / "smoke.jsonl" 

174 event_bus = EventBus(max_queue_size=10_000) 

175 event_logger = EventLogger(log_path=log_path, event_bus=event_bus) 

176 

177 agent_configs = scenario.get_agents( 

178 default_model="scripted-model", 

179 default_provider="anthropic", 

180 ) 

181 missing = {a.agent_id for a in agent_configs} - set(scripts) 

182 if missing: 182 ↛ 183line 182 didn't jump to line 183 because the condition on line 182 was never true

183 raise AssertionError(f"no script provided for {sorted(missing)}") 

184 

185 # One scripted model per agent, chosen by the model name the runner asks for. 

186 # The runner passes agent_config.model straight through, so naming each 

187 # agent's model after its id is enough to route them apart. 

188 for config in agent_configs: 

189 config.model = f"scripted::{config.agent_id}" 

190 

191 def scripted_model_for(model: str, provider: str) -> object: 

192 _ = provider 

193 agent_id = model.removeprefix("scripted::") 

194 return build_scripted_model( 

195 turns=list(scripts[agent_id]), 

196 # Out of script means "nothing left to do", not "broken": park on a 

197 # poll so the round can end on idle instead of the agent dying. 

198 when_exhausted=IDLE_CYCLE, 

199 )

201 def idle_is_enough(round_age: float) -> bool: 

202 """The test's answer to "have the agents finished?": the idle check itself. 

203 

204 An idle agent is blocked on ``read_notifications`` with an empty queue 

205 and nothing in flight, so it cannot act again until something wakes it. 

206 That is a state, not a guess, and needs no duration to confirm it. A run 

207 waits out a floor instead because a real model can be slow between 

208 turns; here nothing is, so the floor only adds a race between the clock 

209 and the agents. See "Tests and time" in CLAUDE.md. 

210 """ 

211 _ = round_age 

212 return True
