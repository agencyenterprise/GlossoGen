1"""Procedural anomaly-case generation for the orbital_anomaly scenario. 

2 

3Defines twelve fault signatures grouped into subsystems and generates one 

4cascading anomaly per round by combining faults with seed-based 

5randomisation. 

6 

7Each fault carries several single-action corrective procedure *variants*, 

8all of which are plausible responses to that fault. Per round a secret 

9``variant_index`` (held only by the Systems Engineer, via the config table) 

10selects which variant applies to every fault this round. Because the variant 

11is one of the fault's own coherent procedures, the choice is unpredictable 

12(so the engineer is needed every round) without being incoherent (so the 

13crew has no intuitive "right answer" to override it with). 

14 

15The live parameter values are split across the two observers: ``unit`` and 

16``config`` are read off the panel by the astronaut, ``hold_seconds`` and 

17``setting`` are read from telemetry by the Telemetry Officer. Faults in the 

18same subsystem share a cockpit alarm and a generic panel template, so the 

19astronaut can read off the affected unit and configuration but cannot tell 

20which fault it is; the telemetry signature disambiguates. 

21""" 

22

309def get_config_variant_mapping(variant_index: int) -> list[ConfigMappingEntry]: 

310 """Build the fault-to-selected-variant table for one round (engineer's view).

333def _build_stage(fault_index: int, variant_index: int, rng: random.Random) -> AnomalyStage: 

334 """Construct one stage by drawing live parameters and applying the selected variant.""" 

335 fault = FAULT_SIGNATURES[fault_index] 

336 unit = rng.choice(_UNITS) 

337 config = rng.choice(_CONFIGS) 

338 hold_seconds = rng.choice(_HOLD_SECONDS) 

339 setting = rng.choice(_SETTINGS) 

340 return AnomalyStage( 

341 fault_name=fault.name, 

342 subsystem=fault.subsystem, 

343 cockpit_alarm=fault.cockpit_alarm, 

344 panel_observation=fault.panel_observation.format(unit=unit, config=config), 

345 telemetry_readout=_render_telemetry_readout( 

346 signature=fault.telemetry_signature, 

347 hold_seconds=hold_seconds, 

348 setting=setting, 

349 ), 

350 judge_expected_actions=fault.procedure_variants[variant_index].format( 

351 unit=unit, 

352 config=config, 

353 hold_seconds=hold_seconds, 

354 setting=setting, 

355 ), 

356 )

359def _select_fault_indices( 

360 rng: random.Random, 

361 round_number: int, 

362 easy_fault_indices: list[int], 

363 easy_round_numbers: frozenset[int], 

364 fault_count_values: list[int], 

365 fault_count_weights: list[int], 

366) -> list[int]: 

367 """Draw fault indices for one round, forcing easy rounds to a single fault. 

368 

369 Always consumes the same RNG calls in the same order so that non-easy 

370 rounds under a given seed reproduce their case content regardless of 

371 which rounds are marked easy. 

372 """ 

373 pool_size = len(FAULT_SIGNATURES) 

374 num_faults = min( 

375 rng.choices(population=fault_count_values, weights=fault_count_weights, k=1)[0], 

376 pool_size, 

377 ) 

378 selected_indices = rng.sample(range(pool_size), k=num_faults) 

379 if round_number in easy_round_numbers: 

380 return [easy_fault_indices[selected_indices[0] % len(easy_fault_indices)]] 

381 return selected_indices

384def get_cases( 

385 seed: int, 

386 round_count: int, 

387 round_time_budget_seconds: int, 

388 cipher_enabled: bool, 

389 easy_round_numbers: frozenset[int], 

390 fault_count_values: list[int], 

391 fault_count_weights: list[int], 

392) -> list[AnomalyCase]: 

393 """Generate one cascading anomaly per round via seed-based selection. 

394 

395 Most rounds draw their fault count from the weighted distribution; rounds 

396 listed in ``easy_round_numbers`` are forced to a single priority-<=2 

397 fault. Every round uses the same fixed time budget. When ``cipher_enabled`` 

398 is false the per-round variant index is forced to zero so every fault uses 

399 its first procedure variant; the index is still drawn from the RNG either 

400 way to keep the case stream deterministic. 

401 """ 

402 rng = random.Random(seed) 

403 easy_fault_indices = [idx for idx, f in enumerate(FAULT_SIGNATURES) if f.priority <= 2] 

404 cases: list[AnomalyCase] = [] 

405 for i in range(round_count): 

406 selected_indices = _select_fault_indices( 

407 rng=rng, 

408 round_number=i + 1, 

409 easy_fault_indices=easy_fault_indices, 

410 easy_round_numbers=easy_round_numbers, 

411 fault_count_values=fault_count_values, 

412 fault_count_weights=fault_count_weights, 

413 ) 

414 priority_order = sorted(selected_indices, key=lambda idx: FAULT_SIGNATURES[idx].priority) 

415 drawn_variant = rng.randrange(NUM_VARIANTS) 

416 if cipher_enabled: 416 ↛ 419line 416 didn't jump to line 419 because the condition on line 416 was always true

417 variant_index = drawn_variant 

418 else: 

419 variant_index = 0 

420 stages = tuple( 

421 _build_stage(fault_index=idx, variant_index=variant_index, rng=rng) 

422 for idx in priority_order 

423 ) 

424 cases.append( 

425 AnomalyCase( 

426 case_number=i + 1, 

427 fault_name=" + ".join(FAULT_SIGNATURES[idx].name for idx in priority_order), 

428 stages=stages, 

429 time_budget_seconds=round_time_budget_seconds, 

430 variant_index=variant_index, 

431 ) 

432 ) 

433 return cases
