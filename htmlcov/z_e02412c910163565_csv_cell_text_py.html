1"""Turning a Python value into one CSV cell. 

2 

3Two problems this solves, both of which come from the cells carrying model output. 

4 

5Control characters. Agent messages and judge summaries contain them, and a 

6spreadsheet refuses a file that does. Stripping beats escaping here: none of them 

7mean anything in a data frame. 

8 

9Embedded newlines. A newline inside a quoted field is legal CSV, and the 

10line-oriented tools people reach for first do not read it that way: `wc -l` 

11overcounts, `head` truncates mid-record, and a naive split lands half a row in the 

12next record. Every field that can contain one holds prose, where a space loses 

13nothing. 

14 

15Leading characters a spreadsheet reads as a formula. A judge note really does 

16begin `@ notation established: '@B' means ...`, and Excel renders a cell starting 

17with `@` as `#NAME?`, losing the note without saying so. Such cells are prefixed 

18with an apostrophe, which Excel treats as "this is text" and does not display. 

19 

20`-` is deliberately not guarded: a leading minus is far more often a negative 

21number than a formula, and numbers do not pass through the text path anyway. 

22 

23Floats keep their full repr. Rounding to a fixed number of decimals here would 

24silently discard precision a metric computed. 

25 

26`nan` is written as an empty cell. The per-message scorers return it for text 

27they cannot score at all, and an empty cell is how this export says no number 

28exists. Writing the token `nan` would put a value in a numeric column that half 

29the readers parse as a string. 

30""" 

31 

32import math 

33import re 

34from datetime import datetime 

35from typing import Any, cast 

36 

37import orjson

39# C0 minus tab, plus DEL and C1. Tab survives because it is a legal cell 

40# character that `csv` quotes correctly. 

41_CONTROL_CHARS_RE = re.compile(r"[\x00-\x08\x0b\x0c\x0e-\x1f\x7f-\x9f]") 

42 

43_NEWLINE_RE = re.compile(r"\r\n|\r|\n") 

44 

45MULTI_VALUE_SEPARATOR = "; " 

46 

47# `-` is absent on purpose: see the module docstring. 

48_FORMULA_LEAD_CHARS = ("=", "+", "@", "\t")

51def guard_spreadsheet_formula(text: str) -> str: 

52 """Prefix an apostrophe when a spreadsheet would read the cell as a formula.""" 

53 if text.startswith(_FORMULA_LEAD_CHARS): 

54 return f"'{text}" 

55 return text

58def sanitize_cell_text(text: str) -> str: 

59 """Strip control characters and flatten newlines to single spaces.""" 

60 without_newlines = _NEWLINE_RE.sub(" ", text) 

61 return _CONTROL_CHARS_RE.sub("", without_newlines)

64def render_cell(text: str) -> str: 

65 """Render a finished text cell: sanitized, then guarded against formula reading. 

66 

67 Applied once per cell, not per fragment: only the first character 

68 of the whole cell decides how a spreadsheet reads it. 

69 """ 

70 return guard_spreadsheet_formula(text=sanitize_cell_text(text=text))

109def render_json(value: Any) -> str: 

110 """Render a container as compact JSON with sorted keys.""" 

111 if isinstance(value, (set, frozenset)): 

112 # A JSON round trip cannot represent a set, and its iteration order is 

113 # not stable across processes, so it is sorted into a list first. 

114 return render_json(value=sorted(cast(frozenset[Any], value), key=str)) 

115 raw = orjson.dumps(value, option=orjson.OPT_SORT_KEYS).decode("utf-8") 

116 return sanitize_cell_text(text=raw)

119def render_string_list(values: list[str]) -> str: 

120 """Join multi-value cells with a separator that does not fight the format. 

121 

122 A comma would need the whole cell quoted and still reads as a column break to 

123 anyone eyeballing the file. 

124 """ 

125 joined = MULTI_VALUE_SEPARATOR.join(sanitize_cell_text(text=value) for value in values) 

126 return guard_spreadsheet_formula(text=joined)
