1"""FastAPI router for cross-run analysis: what a selection carries, and one query. 

2 

3Two POSTs, for the same reason the export endpoints are POSTs: a selection can name 

4hundreds of runs and a query can name a dozen dimensions and measures, which as a 

5query string exceeds what proxies accept and comes back as an opaque 414. 

6 

7A selection that matches nothing is answered rather than refused. A saved dashboard 

8points at a cohort by filters, and a cohort that is empty today is a thing to render 

9as "no runs match" rather than an error to explain. Ids that no longer resolve are 

10reported on the answer for the same reason: a dashboard outlives its runs. 

11 

12The run ceiling is the export's, since the work is the same work: one evaluation 

13report read per selected run. 

14""" 

15

44class SelectedRecords(NamedTuple): 

45 """The loaded runs a selection names, and the ids it named that are gone.""" 

46 

47 records: list[AnalysisRunRecord] 

48 missing_run_ids: list[str] 

49 

50 

51async def _selected_records( 

52 request: Request, 

53 selection: RunSelection, 

54 grain: AnalysisGrain, 

55) -> SelectedRecords: 

56 """Resolve and load the runs a selection names within the active group. 

57 

58 The grain decides whether the metrics' sidecars are read, which is a filesystem 

59 read per metric per run and is why it is not done for the grains that cannot use 

60 them. It is part of the cache key for the same reason: records loaded without 

61 sidecars would answer a keyed query with nothing. 

62 """ 

63 resolved = await resolve_export_selection(request=request, selection=selection) 

64 try: 

65 check_run_count(run_count=len(resolved.summaries)) 

66 except ExportTooLargeError as exc: 

67 raise HTTPException(status_code=413, detail=str(exc)) from exc 

68 

69 identity = get_identity(request=request) 

70 cache = request.app.state.analysis_record_cache 

71 read_sidecars = grain is AnalysisGrain.KEYED 

72 key = f"{identity.active_group_id}|{read_sidecars}|{selection.model_dump_json()}" 

73 

74 async def load() -> list[AnalysisRunRecord]: 

75 """Read every selected run's report, keeping only what a query reads.""" 

76 return await load_analysis_records( 

77 runs=resolved.summaries, read_sidecars=read_sidecars 

78 ) 

79 

80 records = await cache.records( 

81 key=key, 

82 now=time.monotonic(), 

83 run_count=len(resolved.summaries), 

84 load=load, 

85 ) 

86 return SelectedRecords(records=records, missing_run_ids=sorted(resolved.missing_run_ids))

89@router.post("/runs/analysis/fields", response_model=AnalysisFieldCatalog) 

90async def analysis_fields( 

91 body: AnalysisFieldsRequest, 

92 request: Request, 

93) -> AnalysisFieldCatalog: 

94 """Describe what this selection can be grouped, filtered, and measured by.""" 

95 selected = await _selected_records( 

96 request=request, selection=body.selection, grain=body.grain 

97 ) 

98 catalog = build_field_catalog(records=selected.records, grain=body.grain) 

99 return catalog.model_copy(update={"missing_run_ids": selected.missing_run_ids})

102@router.post("/runs/analysis/query", response_model=AnalysisResult) 

103async def analysis_query( 

104 body: AnalysisQueryRequest, 

105 request: Request, 

106) -> AnalysisResult: 

107 """Group and aggregate this selection, one row per group.""" 

108 selected = await _selected_records( 

109 request=request, selection=body.selection, grain=body.query.grain 

110 ) 

111 logger.info( 

112 "Analysis query over %d runs at the %s grain, grouped by %s", 

113 len(selected.records), 

114 body.query.grain.value, 

115 ", ".join(body.query.group_by), 

116 ) 

117 result = run_analysis_query(records=selected.records, spec=body.query) 

118 return result.model_copy(update={"missing_run_ids": selected.missing_run_ids})
