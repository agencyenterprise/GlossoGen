80def list_providers() -> list[Provider]: 

81 """The providers the pickers offer: those of the hosted models, then ``self-hosted``. 

82 

83 ``self-hosted`` is listed only when ``SELF_HOSTED_BASE_URLS`` names at least 

84 one model. This is the picker listing, not the set a run may name: that is 

85 ``SIMULATION_PROVIDERS``. 

86 """ 

87 seen: set[Provider] = set() 

88 providers: list[Provider] = [] 

89 for _, provider in _HOSTED_MODELS: 

90 if provider not in seen: 

91 seen.add(provider) 

92 providers.append(provider) 

93 if _get_self_hosted_model_names(): 

94 providers.append(Provider.SELF_HOSTED) 

95 return providers

104 self_hosted = [(name, Provider.SELF_HOSTED) for name in _get_self_hosted_model_names()] 

105 return list(_HOSTED_MODELS) + self_hosted
