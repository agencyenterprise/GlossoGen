1"""Holding a selection's loaded runs for a short while. 

2 

3Building a chart is a conversation: change the grouping, add a measure, flip an 

4aggregate. Each of those is one query, and each would otherwise re-read one evaluation 

5report per selected run. On a few hundred runs that is the whole cost of the request, 

6repeated for a change that touched no run at all. 

7 

8The budget is counted in runs, not in entries, because that is what the memory is. An 

9entry count would bound sixteen cohorts of any width, which bounds nothing. What is 

10held is the projected record, an order of magnitude smaller than the full report it 

11came from, which is what lets a scenario-wide cohort fit. A selection larger than the 

12whole budget is loaded and not kept, so a very wide cohort is slow to edit rather than 

13fatal to hold. 

14 

15Entries expire quickly, because a run that has just been evaluated should appear 

16without anyone restarting a server. 

17 

18Concurrent requests for one selection share a single load. React Query already 

19deduplicates on the client, but a shared dashboard opened by two people at once does 

20not go through one client. 

21 

22Time is a parameter rather than something read here, so a test states what "later" 

23means instead of waiting for it. 

24"""

48class AnalysisRecordCache:

51 def __init__(self, ttl_seconds: float, max_runs: int) -> None: 

52 self._ttl_seconds = ttl_seconds 

53 self._max_runs = max_runs 

54 self._entries: OrderedDict[str, CacheEntry] = OrderedDict() 

55 

56 def cached_run_count(self) -> int: 

57 """Return how many runs are currently held across every entry.""" 

58 return sum(entry.run_count for entry in self._entries.values()) 

59 

60 def _drop_expired(self, now: float) -> None: 

61 """Remove entries whose window has passed.""" 

62 expired = [key for key, entry in self._entries.items() if entry.expires_at <= now] 

63 for key in expired: 

64 self._entries.pop(key, None) 

65 

66 def _make_room(self, run_count: int) -> None: 

67 """Evict least recently used entries until this many more runs fit.""" 

68 while self._entries and self.cached_run_count() + run_count > self._max_runs: 

69 self._entries.popitem(last=False)
