1"""Provider-specific token counters for measuring message length. 

2 

3Each counter calls its provider's official token counting API to return 

4an exact count. Providers without a counting API fall back to a word-count 

5heuristic. The factory function selects the right implementation based on 

6the provider name. 

7"""

20class TokenCounter(ABC): 

21 """Counts the number of tokens in a text string using a provider's tokenizer. 

22 

23 Results are cached with an LRU eviction policy so repeated calls with the 

24 same text skip the external API round-trip. 

25 """ 

26 

27 def __init__(self) -> None: 

28 self._cache: cachetools.LRUCache[str, int] = cachetools.LRUCache(maxsize=1024) 

29 

30 async def count(self, text: str) -> int: 

31 """Return the token count for the given text, using the cache when possible.""" 

32 cached = self._cache.get(text) 

33 if cached is not None: 

34 return cached 

35 result = await self._count_impl(text=text) 

36 self._cache[text] = result 

37 return result

45class AnthropicTokenCounter(TokenCounter): 

46 """Counts tokens using the Anthropic Messages count_tokens API. 

47 

48 The API is free and returns the exact token count that the model's 

49 tokenizer produces. Requires ``ANTHROPIC_API_KEY`` in the environment. 

50 """ 

51 

52 def __init__(self, model: str) -> None: 

53 super().__init__() 

54 api_key = os.environ.get("ANTHROPIC_API_KEY", "") 

55 self._client = anthropic.AsyncAnthropic(api_key=api_key) 

56 self._model = model 

57 

58 async def _count_impl(self, text: str) -> int: 

59 """Count tokens via the Anthropic count_tokens endpoint.""" 

60 try: 

61 response = await self._client.messages.count_tokens( 

62 model=self._model, 

63 messages=[{"role": "user", "content": text}], 

64 ) 

65 return response.input_tokens 

66 except Exception: 

67 logger.exception( 

68 "Anthropic token count failed for model %s, falling back to word count", 

69 self._model, 

70 ) 

71 return len(text.split())

74class OpenAITokenCounter(TokenCounter): 

75 """Counts tokens using the OpenAI Responses input_tokens.count API. 

76 

77 Returns the exact token count for the model's tokenizer. Requires 

78 ``OPENAI_API_KEY`` in the environment. 

79 """ 

80 

81 def __init__(self, model: str) -> None: 

82 super().__init__() 

83 api_key = os.environ.get("OPENAI_API_KEY", "") 

84 self._client = openai.AsyncOpenAI(api_key=api_key) 

85 self._model = model 

86 

87 async def _count_impl(self, text: str) -> int: 

88 """Count tokens via the OpenAI input_tokens.count endpoint.""" 

89 try: 

90 response = await self._client.responses.input_tokens.count( 

91 model=self._model, 

92 input=text, 

93 ) 

94 return response.input_tokens 

95 except Exception: 

96 logger.exception( 

97 "OpenAI token count failed for model %s, falling back to word count", 

98 self._model, 

99 ) 

100 return len(text.split())

111def create_token_counter(provider: str, model: str) -> TokenCounter: 

112 """Create the appropriate token counter for a given provider and model. 

113 

114 Returns an ``AnthropicTokenCounter`` for Anthropic, an ``OpenAITokenCounter`` 

115 for OpenAI, or a ``HeuristicTokenCounter`` for all other providers. 

116 """ 

117 if provider == "anthropic": 

118 return AnthropicTokenCounter(model=model) 

119 if provider == "openai": 

120 return OpenAITokenCounter(model=model) 

121 logger.warning( 

122 "No token counting API for provider %s, using word-count heuristic", 

123 provider, 

124 ) 

125 return HeuristicTokenCounter()
