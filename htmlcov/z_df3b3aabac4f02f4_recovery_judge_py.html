1"""LLM-based judge that evaluates whether a floor associate's recovery action 

2satisfies every required criterion for the current warehouse robot case. 

3 

4Uses ``generate_structured`` to get a validated per-criterion verdict with 

5a free-form explanation from the configured judge model. 

6""" 

7 

8import logging 

9from pathlib import Path 

10 

11from pydantic import BaseModel 

12 

13from glossogen.llm.provider import LLMMessage, LLMProvider 

14from glossogen.scenarios.warehouse_robot_recovery.events import WarehouseRecoveryJudgment 

15from glossogen.template_renderer import TemplateRenderer 

16 

17logger = logging.getLogger(__name__) 

18 

19PROMPTS_DIR = Path(__file__).parent / "prompts" 

20_renderer = TemplateRenderer(prompts_dirs=[PROMPTS_DIR]) 

21 

22 

23class RecoveryJudgmentResult(BaseModel): 

24 """Structured output from the warehouse recovery judge LLM call.""" 

25 

26 judgment: WarehouseRecoveryJudgment 

27 explanation: str 

28 

29 

30async def judge_recovery( 

31 provider: LLMProvider, 

32 robot_id: str, 

33 expected_procedure: str, 

34 safety_constraints: str, 

35 observer_action: str, 

36) -> RecoveryJudgmentResult: 

37 """Ask the LLM judge whether the recovery action meets every required criterion. 

38 

39 ``expected_procedure`` is the fully rendered ordered list of recovery 

40 steps for the round's faults (parameters already substituted). 

41 ``safety_constraints`` lists the round's forbidden actions and 

42 constraints from the safety coordinator. The judge returns a 

43 per-criterion ``WarehouseRecoveryJudgment`` plus a free-form 

44 explanation. 

45 """ 

46 system_prompt = _renderer.render( 

47 template_name="recovery_judge.jinja", 

48 template_variables={}, 

49 ) 

50 user_message = ( 

51 f"Target robot: {robot_id}\n\n" 

52 f"Expected recovery procedure (in required order):\n{expected_procedure}\n\n" 

53 f"Safety constraints:\n{safety_constraints}\n\n" 

54 f"Floor associate's reported action:\n{observer_action}" 

55 ) 

56 logger.info( 

57 "Recovery judge input: robot=%s expected=[%s] safety=[%s] action=[%s]", 

58 robot_id, 

59 expected_procedure, 

60 safety_constraints, 

61 observer_action, 

62 ) 

63 result = await provider.generate_structured( 

64 system_prompt=system_prompt, 

65 messages=[LLMMessage(role="user", content=user_message)], 

66 output_schema=RecoveryJudgmentResult, 

67 ) 

68 judgment = result.judgment 

69 logger.info( 

70 "Recovery judge result: robot=%s judgment=%s explanation=%s", 

71 robot_id, 

72 judgment.model_dump(), 

73 result.explanation, 

74 ) 

75 return result
