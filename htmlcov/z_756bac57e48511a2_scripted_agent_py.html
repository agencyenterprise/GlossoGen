1"""A pydantic-ai model whose turns are written by the test. 

2 

3Agents normally decide what to do by calling a provider. Here the decision is a 

4list: turn N of the script is what the agent does on its Nth cycle. Runs become 

5deterministic and free, so a test can assert what an agent did instead of hoping 

6an LLM chose it. 

7 

8Built on pydantic-ai's ``FunctionModel``, which slots into the same 

9``build_pydantic_ai_model`` seam the real models use. Everything below the model 

10stays real: tools, the MCP toolset, the runtime, the event log. 

11""" 

12 

13import json 

14from collections.abc import AsyncIterator, Sequence 

15from dataclasses import dataclass 

16 

17from pydantic_ai.messages import ModelMessage, ModelResponse, TextPart, ToolCallPart 

18from pydantic_ai.models.function import AgentInfo, DeltaToolCall, DeltaToolCalls, FunctionModel 

19 

20 

21@dataclass(frozen=True) 

22class ToolTurn: 

23 """One turn in which the agent calls a tool.""" 

24 

25 tool_name: str 

26 args: dict[str, object] 

27 

28 

29@dataclass(frozen=True) 

30class SayTurn: 

31 """One turn in which the agent replies with text and stops.""" 

32 

33 text: str 

34 

35 

36ScriptedTurn = ToolTurn | SayTurn 

37 

38 

39class ScriptExhausted(RuntimeError): 

40 """Raised when an agent takes more turns than the script provides. 

41 

42 A silent fallback would let a test pass while the agent looped somewhere the 

43 script never described. 

44 """ 

45 

46 

47def build_scripted_model( 

48 *, 

49 turns: Sequence[ScriptedTurn], 

50 when_exhausted: Sequence[ScriptedTurn] | None, 

51) -> FunctionModel: 

52 """Return a model that plays ``turns`` in order, one per model call. 

53 

54 A turn is one model call, not one round. The runner wakes an agent, lets it 

55 act until it answers with text, then parks it until the next notification. A 

56 script covering a whole simulation needs turns for every one of those wakes, 

57 not just the first. 

58 

59 ``when_exhausted`` decides what happens once the script runs out. ``None`` 

60 raises, which is what a component test wants: an agent looping past its 

61 script is a bug and a silent fallback would hide it. A simulation instead 

62 passes the cycle an idle agent repeats, usually a poll and a reply, so the 

63 run can reach its own end instead of dying mid-round. 

64 

65 Both a plain and a streaming implementation are supplied. The agent runner 

66 streams, so the streaming one is what actually runs; the plain one keeps the 

67 fake usable from a direct ``agent.run()``. 

68 """ 

69 remaining = list(turns) 

70 idle_cycle = list(when_exhausted) if when_exhausted is not None else [] 

71 

72 def take_turn() -> ScriptedTurn: 

73 if not remaining: 

74 if not idle_cycle: 

75 raise ScriptExhausted( 

76 f"agent asked for turn {len(turns) + 1} but the script has {len(turns)}" 

77 ) 

78 remaining.extend(idle_cycle) 

79 return remaining.pop(0) 

80 

81 def next_turn(messages: list[ModelMessage], info: AgentInfo) -> ModelResponse: 

82 _ = messages 

83 _ = info 

84 turn = take_turn() 

85 if isinstance(turn, SayTurn): 

86 return ModelResponse(parts=[TextPart(content=turn.text)]) 

87 return ModelResponse(parts=[ToolCallPart(tool_name=turn.tool_name, args=dict(turn.args))]) 

88 

89 async def stream_turn( 

90 messages: list[ModelMessage], info: AgentInfo 

91 ) -> AsyncIterator[str | DeltaToolCalls]: 

92 _ = messages 

93 _ = info 

94 turn = take_turn() 

95 if isinstance(turn, SayTurn): 

96 yield turn.text 

97 return 

98 # One delta carrying the whole call. The runner only cares that the 

99 # call arrives, not that it was split across chunks. 

100 deltas: DeltaToolCalls = { 

101 0: DeltaToolCall(name=turn.tool_name, json_args=json.dumps(dict(turn.args))) 

102 } 

103 yield deltas 

104 

105 return FunctionModel(next_turn, stream_function=stream_turn)
