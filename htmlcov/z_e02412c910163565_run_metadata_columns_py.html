1"""The columns describing a run itself, before knobs or scores. 

2 

3A curated projection of the run summary, not every field on it. The omissions are 

4deliberate: the scenario description is prose that repeats identically on every row 

5of a single-scenario export, and the evaluation content hash addresses a cache and 

6says nothing about the run. 

7 

8`run_id` and `scenario_name` are always emitted and cannot be deselected. Every 

9other frame joins back on `run_id`, and a CSV whose rows cannot be identified is 

10not worth writing. 

11""" 

12 

13from glossogen.run_export.csv_cell_text import render_cell, render_scalar, render_string_list 

14from glossogen.run_export.model_weight_class import MODEL_CLASS_COLUMN, model_class_of 

15from glossogen.server.runs.models import RunSummary 

16 

17RUN_ID_COLUMN = "run_id" 

18SCENARIO_NAME_COLUMN = "scenario_name" 

19 

20IDENTITY_COLUMNS: tuple[str, ...] = (RUN_ID_COLUMN, SCENARIO_NAME_COLUMN) 

21 

22 

23def run_metadata_cells(summary: RunSummary) -> dict[str, str]: 

24 """Return the identity and metadata cells for one run.""" 

25 return { 

26 RUN_ID_COLUMN: render_cell(text=summary.run_id), 

27 SCENARIO_NAME_COLUMN: render_cell(text=summary.scenario_name), 

28 "run_dir_name": render_cell(text=summary.run_id.split("/", 1)[-1]), 

29 "timestamp": summary.timestamp.isoformat(), 

30 "status": render_cell(text=summary.status.value), 

31 "current_round": render_scalar(value=summary.current_round), 

32 "total_messages": render_scalar(value=summary.total_messages), 

33 "total_cost_usd": render_scalar(value=summary.total_cost_usd), 

34 "duration_seconds": render_scalar(value=summary.duration_seconds), 

35 "provider": render_cell(text=summary.provider), 

36 MODEL_CLASS_COLUMN: model_class_of(agent_models=summary.agent_models), 

37 "models": render_string_list(values=summary.models), 

38 "labels": render_string_list(values=summary.labels), 

39 "has_evaluation": render_scalar(value=summary.has_evaluation), 

40 "has_note": render_scalar(value=summary.has_note), 

41 "run_dir": render_cell(text=summary.run_dir), 

42 }
