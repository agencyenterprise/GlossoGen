1"""The type of a manifest field that names a file inside the manifest's own run directory. 

2 

3Readers join such a field onto the run directory and open the result. A manifest can 

4arrive in an uploaded bundle, so the field is refused unless it is a bare file name: 

5a path with a separator, an absolute path or ``..`` would let the uploader choose a 

6file in another run, including another group's on a shared runs directory. 

7""" 

8 

9from typing import Annotated 

10 

11from pydantic import AfterValidator 

12 

13 

14def _require_bare_file_name(value: str) -> str: 

15 """Return ``value`` if it names a file directly inside a directory, else raise.""" 

16 if value in ("", ".", "..") or "/" in value or "\\" in value or "\0" in value: 

17 raise ValueError(f"must be a file name inside the run directory, got {value!r}") 

18 return value 

19 

20 

21RunDirFileName = Annotated[str, AfterValidator(_require_bare_file_name)] 

22"""A file name with no directory part, read relative to the run directory."""
