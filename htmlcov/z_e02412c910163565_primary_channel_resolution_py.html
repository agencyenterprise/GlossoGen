1"""Which of a run's channels the scenario considers primary, and whose team it is. 

2 

3The message table needs this for two columns a hand-written exporter hardcodes: 

4whether a message was on the budgeted task channel, and which team sent it. 

5`get_primary_channels` is a required scenario hook, so asking the scenario is 

6scenario-agnostic in a way naming channel ids here would not be. The scenario is 

7rebuilt from the run's recorded config by `recorded_scenario_rebuild`. 

8 

9When nothing rebuilds it, the two columns render empty, which says "not known" 

10rather than "not primary". 

11""" 

12 

13import logging 

14from typing import Any, NamedTuple 

15 

16from glossogen.recorded_scenario_rebuild import rebuild_recorded_scenario 

17from glossogen.scenario_protocol import SimulationScenario 

18 

19logger = logging.getLogger(__name__) 

20 

21 

22class PrimaryChannelMap(NamedTuple): 

23 """The primary channels of one run, and the team each belongs to. 

24 

25 ``resolved`` is False when the scenario could not be rebuilt, in which case 

26 ``team_by_channel`` is empty and says nothing about any channel. 

27 ``team_by_channel`` maps a primary channel id to its team id, empty string 

28 for a single-team scenario. 

29 """ 

30 

31 resolved: bool 

32 team_by_channel: dict[str, str] 

33 

34 

35UNRESOLVED = PrimaryChannelMap(resolved=False, team_by_channel={}) 

36 

37 

38def _team_by_channel(scenario: SimulationScenario) -> dict[str, str]: 

39 """Map each of the scenario's primary channel ids to its team id.""" 

40 team_by_channel: dict[str, str] = {} 

41 for channel in scenario.get_primary_channels(): 

42 team_id = "" 

43 if channel.team_id is not None: 

44 team_id = channel.team_id 

45 team_by_channel[channel.channel_id] = team_id 

46 return team_by_channel 

47 

48 

49def resolve_primary_channels( 

50 scenario_name: str, 

51 scenario_config: dict[str, Any], 

52) -> PrimaryChannelMap: 

53 """Return the run's primary channels, or ``UNRESOLVED`` when nothing rebuilds it.""" 

54 scenario = rebuild_recorded_scenario( 

55 scenario_name=scenario_name, 

56 scenario_config=scenario_config, 

57 ) 

58 if scenario is None: 

59 logger.info("Exporting %s messages without primary-channel or team columns", scenario_name) 

60 return UNRESOLVED 

61 return PrimaryChannelMap(resolved=True, team_by_channel=_team_by_channel(scenario=scenario))
