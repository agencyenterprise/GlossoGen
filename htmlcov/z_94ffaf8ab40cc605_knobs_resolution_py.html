1"""Resolve what `--config` and `--knobs` name into scenario knobs. 

2 

3A scenario ships its presets inside its own package, so a path to one is only 

4typeable by someone who can see that package. From a checkout that is 

5`src/glossogen/scenarios/<name>/knobs_default.json`; from an installed 

6distribution it is a path into `site-packages`, which nobody should have to 

7write. Scenarios already publish their presets by name for the API and the MCP 

8tools, so the CLI resolves a name too, and both layouts spell the same command. 

9 

10A file still wins when the argument is one, which is how an experiment keeps its 

11own knobs JSON outside any package. 

12 

13Neither flag falls back to a preset of its own accord. `--config` is required, so 

14a run's configuration is stated rather than inferred: the JSONL records what a 

15run was launched with, and a configuration nobody chose is one nobody can 

16account for later. 

17"""

32 

33 ``source`` is for the log, so a run says which preset or file it read.

41 scenario_cls: type[SimulationScenario],

44 """Return the knobs `--config` names, a preset or a file. 

45 

46 Raises ``SystemExit`` when the argument is neither a file nor a preset the

48 """ 

49 return _resolve_file_or_preset(scenario_cls=scenario_cls, requested=requested, flag="--config")

52def resolve_knobs_overrides( 

53 scenario_cls: type[SimulationScenario], 

54 requested: str | None, 

55) -> ResolvedKnobs | None: 

56 """Return the overrides `--knobs` names, or None when it was not given. 

57 

58 Same vocabulary as ``--config``, a file or a preset name, so one rule covers 

59 both flags. What differs is that this one is optional: no overrides is a 

60 resumed run's normal state. 

61 

62 A preset used this way replaces every field it declares, which is what a 

63 whole-preset JSON file passed here has always done. 

64 """ 

65 if requested is None: 

66 return None 

67 return _resolve_file_or_preset(scenario_cls=scenario_cls, requested=requested, flag="--knobs")

70def _resolve_file_or_preset( 

71 scenario_cls: type[SimulationScenario], 

72 requested: str, 

73 flag: str, 

74) -> ResolvedKnobs: 

75 """Read ``requested`` as a file if it is one, else as a preset name.""" 

76 path = Path(requested) 

77 if path.is_file(): 

78 return ResolvedKnobs( 

79 config=_require_object(payload=orjson.loads(path.read_bytes()), source=str(path)), 

80 source=str(path), 

81 ) 

82 

83 available = scenario_cls.knobs_preset_names() 

84 name = requested.removesuffix(".json") 

85 if name in available: 

86 return ResolvedKnobs( 

87 config=scenario_cls.load_knobs_preset(preset_name=name), 

88 source=f"preset {name!r}", 

89 ) 

90 

91 raise SystemExit( 

92 f"{flag} {requested!r} is neither a readable file nor a preset " 

93 f"{scenario_cls.name()} ships. Its presets: {', '.join(available)}." 

94 )

97def _require_object(payload: Any, source: str) -> dict[str, Any]: 

98 """Return the payload as a knobs mapping, or refuse it by name.""" 

99 if not isinstance(payload, dict): 

100 raise SystemExit(f"{source} must contain a JSON object of scenario knobs.") 

101 return {str(key): value for key, value in cast(dict[Any, Any], payload).items()}
