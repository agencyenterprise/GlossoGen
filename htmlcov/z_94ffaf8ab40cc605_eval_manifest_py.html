1"""Discovery file for detecting an in-progress evaluation subprocess. 

2 

3When an evaluation starts (via CLI or web UI), it writes an ``eval_in_progress.json`` 

4manifest into the run directory. The web server checks this file to report evaluation 

5status. The manifest is deleted when the evaluation finishes or crashes. 

6""" 

7 

8import logging 

9import os 

10from pathlib import Path 

11 

12import orjson 

13from pydantic import BaseModel 

14 

15logger = logging.getLogger(__name__) 

16 

17EVAL_MANIFEST_FILENAME = "eval_in_progress.json" 

18 

19 

20class EvalManifest(BaseModel): 

21 """Describes a running evaluation subprocess.""" 

22 

23 pid: int 

24 

25 

26def write_eval_manifest(run_dir: Path, pid: int) -> None: 

27 """Write the evaluation manifest to the run directory.""" 

28 path = run_dir / EVAL_MANIFEST_FILENAME 

29 manifest = EvalManifest(pid=pid) 

30 data = orjson.dumps(manifest.model_dump(mode="json")) 

31 path.write_bytes(data) 

32 logger.info("Wrote eval manifest: %s (pid=%d)", path, pid) 

33 

34 

35def read_eval_manifest(run_dir: Path) -> EvalManifest | None: 

36 """Read the evaluation manifest from a run directory. 

37 

38 Returns None if the file does not exist or if the referenced process 

39 is no longer alive (stale manifest). 

40 """ 

41 path = run_dir / EVAL_MANIFEST_FILENAME 

42 if not path.exists(): 

43 return None 

44 

45 raw = orjson.loads(path.read_bytes()) 

46 manifest = EvalManifest.model_validate(raw) 

47 

48 if not _is_process_alive(pid=manifest.pid): 

49 logger.debug("Stale eval manifest (PID %d dead): %s", manifest.pid, path) 

50 return None 

51 

52 return manifest 

53 

54 

55def delete_eval_manifest(run_dir: Path) -> None: 

56 """Remove the evaluation manifest from the run directory if it exists.""" 

57 path = run_dir / EVAL_MANIFEST_FILENAME 

58 if path.exists(): 

59 path.unlink() 

60 logger.info("Deleted eval manifest: %s", path) 

61 

62 

63def _is_process_alive(pid: int) -> bool: 

64 """Check whether a process with the given PID is still running.""" 

65 try: 

66 os.kill(pid, 0) 

67 except ProcessLookupError: 

68 return False 

69 except PermissionError: 

70 return True 

71 return True
