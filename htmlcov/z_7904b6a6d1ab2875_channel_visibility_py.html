1"""Per-channel history visibility for an agent swapped into a run. 

2 

3A discriminated union (``Full``, ``None``, ``FromRound``) rather than a binary 

4blocked-channels set, so a swap can declare e.g. "show ``link`` only from round 16 

5onwards" without affecting other channels. Swap configs in scenario knobs and the 

6``AgentSwappedMidRun`` event both carry it. 

7""" 

8 

9from typing import Annotated, Literal 

10 

11from pydantic import BaseModel, ConfigDict, Discriminator, Field 

12 

13 

14class ChannelVisibilityFull(BaseModel): 

15 """Channel history fully visible to the swapped-in agent. 

16 

17 The new agent's ``read_channel`` returns the entire prior history 

18 of this channel, and the predecessor's tool calls targeting it 

19 survive history reconstruction unchanged. 

20 """ 

21 

22 model_config = ConfigDict(extra="forbid") 

23 kind: Literal["full"] = "full" 

24 

25 

26class ChannelVisibilityNone(BaseModel): 

27 """Channel history hidden from the swapped-in agent. 

28 

29 The new agent's ``member_join_index`` for the channel is bumped to 

30 the current message count, so ``read_channel`` returns only post-swap 

31 messages. Every send/read tool call the predecessor made on this 

32 channel is dropped from the reconstructed history along with its 

33 matching tool return. 

34 """ 

35 

36 model_config = ConfigDict(extra="forbid") 

37 kind: Literal["none"] = "none" 

38 

39 

40class ChannelVisibilityFromRound(BaseModel): 

41 """Channel history windowed from a specific round onward. 

42 

43 Sets ``member_join_index`` for the swapped-in agent to the channel's 

44 message count at the start of round ``round_floor``. The predecessor's 

45 reconstructed tool calls on this channel are filtered: every 

46 ``read_channel`` call is dropped (its tool-return blob would otherwise 

47 leak older messages), and every ``send_message`` call is dropped iff 

48 its ``ToolCallInvoked.round_number < round_floor``. 

49 """ 

50 

51 model_config = ConfigDict(extra="forbid") 

52 kind: Literal["from_round"] = "from_round" 

53 round_floor: int = Field(ge=1) 

54 

55 

56ChannelVisibility = Annotated[ 

57 ChannelVisibilityFull | ChannelVisibilityNone | ChannelVisibilityFromRound, 

58 Discriminator("kind"), 

59]
