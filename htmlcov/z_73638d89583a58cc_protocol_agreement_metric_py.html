1"""Metric that finds how the agents came to share each convention, and which messages show it. 

2 

3One judge call reads the whole run as a cited transcript (see ``cited_transcript``). 

4Each step the judge returns keeps only the references that exist in the transcript, 

5and a step left with none is dropped. The result goes to the run's 

6``protocol_agreement.json`` sidecar, which the run page shows, and to a measurement 

7whose score counts the conventions that every agent used. 

8"""

89def build_report( 

90 judged: ProtocolAgreementJudgeOutput, entries: list[CitedEntry] 

91) -> ProtocolAgreementReport: 

92 """Turn the judge's references into evidence, keeping only those that exist in the run.""" 

93 entry_by_reference = {entry.reference: entry for entry in entries} 

94 conventions = [ 

95 _convention(judged=convention, entry_by_reference=entry_by_reference) 

96 for convention in judged.conventions 

97 ] 

98 return ProtocolAgreementReport( 

99 summary=judged.summary, 

100 conventions=[convention for convention in conventions if convention.steps], 

101 )

104def _convention( 

105 judged: JudgedConvention, entry_by_reference: dict[str, CitedEntry] 

106) -> ProtocolConvention: 

107 """One convention with only the steps that cite at least one real entry.""" 

108 steps: list[AgreementStep] = [] 

109 for judged_step in judged.steps: 

110 step = _step(judged=judged_step, entry_by_reference=entry_by_reference) 

111 if step is not None: 

112 steps.append(step) 

113 return ProtocolConvention( 

114 name=judged.name, 

115 meaning=judged.meaning, 

116 steps=steps, 

117 adopted_by_all=judged.adopted_by_all, 

118 first_shared_round=judged.first_shared_round, 

119 )

122def _step(judged: JudgedStep, entry_by_reference: dict[str, CitedEntry]) -> AgreementStep | None: 

123 """The step with its cited entries as evidence, or ``None`` when it cites none that exist.""" 

124 evidence: list[EvidenceMessage] = [] 

125 for reference in judged.references: 

126 entry = entry_by_reference.get(reference.strip().strip("[]")) 

127 if entry is None: 

128 logger.warning( 

129 "protocol_agreement: the judge cited %r, which is not in the run", reference 

130 ) 

131 continue 

132 evidence.append( 

133 EvidenceMessage( 

134 message_id=entry.message_id, 

135 round_number=entry.round_number, 

136 agent_id=entry.agent_id, 

137 text=entry.text, 

138 ) 

139 ) 

140 if not evidence: 

141 return None 

142 return AgreementStep( 

143 stage=judged.stage, 

144 round_number=evidence[0].round_number, 

145 agent_id=evidence[0].agent_id, 

146 explanation=judged.explanation, 

147 evidence=evidence, 

148 )

159def measurement_from_report(metric_name: str, report: ProtocolAgreementReport) -> Measurement: 

160 """Score the conventions every agent used; note each step under the round it happened in.""" 

161 notes = [ 

162 RoundNote( 

163 round_number=step.round_number, 

164 note=f"{convention.name}, {step.stage.value} by {step.agent_id}: {step.explanation}", 

165 ) 

166 for convention in report.conventions 

167 for step in convention.steps 

168 ] 

169 shared = [convention for convention in report.conventions if convention.adopted_by_all] 

170 summary = ( 

171 f"{len(shared)} of {len(report.conventions)} conventions were used by every agent. " 

172 f"{report.summary} Steps and their messages are in {PROTOCOL_AGREEMENT_FILE_NAME}." 

173 ) 

174 return Measurement( 

175 metric_name=metric_name, 

176 score=float(len(shared)), 

177 score_unit="conventions used by every agent", 

178 summary=summary, 

179 per_round=merge_round_notes(notes=notes), 

180 per_agent=[], 

181 )
