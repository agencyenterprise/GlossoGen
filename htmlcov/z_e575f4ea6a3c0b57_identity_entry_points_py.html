1"""Identity providers contributed by other installed distributions. 

2 

3A distribution advertises a provider by declaring an entry point, under a group 

4naming the contract version it was written against:: 

5 

6 [project.entry-points."glossogen.identity_provider.v1"] 

7 my_provider = "my_auth.identity_provider:MyIdentityProvider" 

8 

9Reading imports nothing, the same as :mod:`glossogen.scenario_entry_points`. Loading 

10the class is a separate, explicit step; see 

11:mod:`glossogen.server.identity.identity_provider_loader`. 

12""" 

13 

14import re 

15from importlib.metadata import EntryPoint 

16from typing import NamedTuple 

17 

18from glossogen.plugin_entry_points import all_entry_points, by_name 

19from glossogen.server.identity.identity_api import IDENTITY_API_VERSION 

20 

21IDENTITY_ENTRY_POINT_GROUP_PREFIX = "glossogen.identity_provider.v" 

22IDENTITY_ENTRY_POINT_GROUP = f"{IDENTITY_ENTRY_POINT_GROUP_PREFIX}{IDENTITY_API_VERSION}" 

23 

24# The group an author most plausibly types by mistake: the prefix without its 

25# version. Nothing reads it, so a declaration there would vanish unexplained. 

26_UNVERSIONED_GROUP = IDENTITY_ENTRY_POINT_GROUP_PREFIX.removesuffix(".v") 

27_VERSIONED_GROUP = re.compile(rf"^{re.escape(IDENTITY_ENTRY_POINT_GROUP_PREFIX)}\d+$") 

28 

29 

30class IdentityProviderDeclarations(NamedTuple): 

31 """Every installed provider declaration, split by whether this platform reads it. 

32 

33 ``current`` holds what can be resolved. ``other_groups`` maps a name to the group 

34 it was declared under instead, which turns an unreadable declaration from an 

35 absence into something explainable. For an authentication contract that 

36 distinction decides whether the server may boot at all. 

37 """ 

38 

39 current: dict[str, EntryPoint] 

40 other_groups: dict[str, str] 

41 

42 

43def _is_identity_group(group: str) -> bool: 

44 """Return whether a group name is one a provider could plausibly be declared under. 

45 

46 The bare prefix counts, because the group name is the one thing a plug-in author 

47 types by hand and dropping the ``.v1`` is the likeliest single mistake. 

48 """ 

49 return group == _UNVERSIONED_GROUP or _VERSIONED_GROUP.match(group) is not None 

50 

51 

52def identity_provider_declarations() -> IdentityProviderDeclarations: 

53 """Return both halves from a single read of installed metadata.""" 

54 installed = all_entry_points() 

55 current: dict[str, EntryPoint] = {} 

56 other_groups: dict[str, str] = {} 

57 for group in installed.groups: 

58 if not _is_identity_group(group=group): 58 ↛ 59line 58 didn't jump to line 59 because the condition on line 58 was never true

59 continue 

60 declared = by_name(entry_points=installed.select(group=group), group=group) 

61 if group == IDENTITY_ENTRY_POINT_GROUP: 

62 current = declared 

63 continue 

64 for name in declared: 

65 other_groups[name] = group 

66 # A name declared under both a readable group and an unreadable one is readable, 

67 # so it is not a problem to report. That combination is what a half-finished 

68 # migration looks like: the new group added, the old one not yet removed. 

69 for name in current: 

70 other_groups.pop(name, None) 

71 return IdentityProviderDeclarations(current=current, other_groups=other_groups)
