1"""Assembling the requested frames, and delivering them as a zip or a directory. 

2 

3A multi-frame export is a zip carrying a `columns.csv` legend alongside the 

4tables. The legend records each column's family and how many runs filled it, 

5which is the recoverable half of a blank cell: a knob a scenario never declared 

6and a knob it declared as null both render empty, and only the legend says which 

7columns were sparse. 

8 

9It covers the structural columns of each table it wrote, not only the ones a 

10caller could pick. Those carry no coverage count, since they are emitted on every 

11row of their table by construction, but they do carry the units: nothing else in 

12the export says that `character_entropy_bits` is bits per character or that 

13`repetition_factor` counts encodings per information unit. 

14 

15A single-frame export is the bare CSV, so a double-click opens it. It has no 

16legend, which is the tradeoff a caller accepts by asking for one table. 

17"""

114def _structural_rows(request: CsvExportRequest) -> list[list[str]]: 

115 """Describe the columns each requested table emits on every one of its rows. 

116 

117 ``runs_with_value`` is left empty rather than set to the run count: these are 

118 not filled per run, so a coverage number would be answering a question nobody 

119 asked of them. 

120 """ 

121 rows: list[list[str]] = [] 

122 seen: set[str] = set() 

123 for frame in request.frames: 

124 for column in _STRUCTURAL_COLUMNS.get(frame, ()): 

125 if column in seen: 125 ↛ 126line 125 didn't jump to line 126 because the condition on line 125 was never true

126 continue 

127 seen.add(column) 

128 rows.append([column, frame.value, MESSAGE_COLUMN_UNITS.get(column, ""), "", ""]) 

129 return rows

132def build_legend_frame(records: list[ExportRunRecord], request: CsvExportRequest) -> CsvFrame: 

133 """Build the legend naming each column the export wrote, its family, and its coverage.""" 

134 preview = build_export_preview(records=records, missing_run_ids=[], raw_bytes_estimate=None) 

135 requested_columns = set(request.columns) 

136 requested_metrics = set(request.metrics) 

137 

138 rows: list[list[str]] = [] 

139 for column in preview.columns: 

140 if not column.always_included and column.key not in requested_columns: 

141 continue 

142 rows.append( 

143 [column.key, column.group, "", str(column.runs_with_value), str(preview.run_count)] 

144 ) 

145 for metric in preview.metrics: 

146 if metric.metric_name not in requested_metrics: 146 ↛ 147line 146 didn't jump to line 147 because the condition on line 146 was never true

147 continue 

148 rows.append( 

149 [ 

150 f"metric.{metric.metric_name}", 

151 "metric", 

152 metric.score_unit, 

153 str(metric.runs_with_value), 

154 str(preview.run_count), 

155 ] 

156 ) 

157 

158 rows.extend(_structural_rows(request=request)) 

159 

160 return CsvFrame( 

161 name="columns", 

162 header=["column", "group", "unit", "runs_with_value", "run_count"], 

163 rows=iter(rows), 

164 )

167def write_frames_to_zip( 

168 frames: list[CsvFrame], 

169 legend: CsvFrame, 

170 destination: IO[bytes], 

171 check: Callable[[], None] | None, 

172) -> None: 

173 """Write every frame plus the legend into one zip. 

174 

175 ``check`` is called periodically during the write and is expected to raise if the 

176 export has grown too large. ``None`` writes without a ceiling. 

177 """ 

178 with zipfile.ZipFile(destination, mode="w", compression=zipfile.ZIP_DEFLATED) as archive: 

179 for frame in list(frames) + [legend]: 

180 with archive.open(f"{frame.name}.csv", mode="w") as entry: 

181 row_count, frame_bytes = write_frame(frame=frame, destination=entry, check=check) 

182 logger.info("Wrote %s.csv (%d rows, %d bytes)", frame.name, row_count, frame_bytes)

195def write_frames_to_directory( 

196 frames: list[CsvFrame], 

197 legend: CsvFrame, 

198 out_dir: Path, 

199) -> list[Path]: 

200 """Write every frame plus the legend into ``out_dir``; return the written paths.""" 

201 out_dir.mkdir(parents=True, exist_ok=True) 

202 written: list[Path] = [] 

203 for frame in list(frames) + [legend]: 

204 path = out_dir / f"{frame.name}.csv" 

205 with path.open("wb") as handle: 

206 row_count, _ = write_frame(frame=frame, destination=handle, check=None) 

207 logger.info("Wrote %s (%d rows)", path, row_count) 

208 written.append(path) 

209 return written
