1"""Reading a run's per-metric scores out of its evaluation report. 

2 

3Every completed run's summary carries these scores, and some listings build a 

4summary for every run in a group (MCP ``list_runs``, the exports, the status 

5filter). Over 4,738 local runs, reading every report took such a listing from 

61.4s to 2.0s. The scores are therefore held per report, keyed on its modification 

7time and size, so a run whose report has not changed costs one ``stat`` and a 

8repeat listing takes 1.6s. ``write_report`` replaces a report through 

9``os.replace``, so a rewritten report presents a new key unless the new file has 

10the same size and the same modification time as the old one. 

11""" 

12 

13import logging 

14import threading 

15from collections import OrderedDict 

16from pathlib import Path 

17from typing import NamedTuple 

18 

19from pydantic import BaseModel, ValidationError

23logger = logging.getLogger(__name__)

25# A group with more evaluated runs than this re-reads every report on each 

26# listing that covers all of them, since the least recently read entries are 

27# evicted as the listing proceeds. 

28MAX_CACHED_REPORTS = 10_000

39 

40 ``has_report`` is whether the report file exists. ``scores`` holds the 

41 report's measurements in report order; it is empty when there is no report 

42 and None when the report exists but could not be read or parsed. 

43 """

49class _CachedScores(NamedTuple): 

50 """One report's scores and the modification time and size of the file they came from."""

52 modified_ns: int 

53 size: int

57_CACHE: OrderedDict[Path, _CachedScores] = OrderedDict() 

58_CACHE_LOCK = threading.Lock()

61def read_evaluation_scores(report_path: Path) -> EvaluationReportScores: 

62 """Read the report's scores through the cache. 

63 

64 A report that cannot be read or parsed yields ``scores=None`` and one logged 

65 exception per version of the file while that version stays cached, so one 

66 bad report costs that run its scores rather than failing the listing. A 

67 ``stat`` that fails for a reason other than the file being absent is logged 

68 and reported as no report. Called from worker threads. 

69 """ 

70 try: 

71 stat = report_path.stat() 

72 except FileNotFoundError: 

73 return _no_report() 

74 except OSError: 

75 logger.exception("Failed to stat evaluation report %s", report_path) 

76 return _no_report() 

77 with _CACHE_LOCK: 

78 cached = _CACHE.get(report_path) 

79 if ( 

80 cached is not None 

81 and cached.modified_ns == stat.st_mtime_ns 

82 and cached.size == stat.st_size 

83 ): 

84 _CACHE.move_to_end(report_path) 

85 return _as_result(scores=cached.scores) 

86 scores = _parse_scores(report_path=report_path) 

87 with _CACHE_LOCK: 

88 _CACHE[report_path] = _CachedScores( 

89 modified_ns=stat.st_mtime_ns, size=stat.st_size, scores=scores 

90 ) 

91 _CACHE.move_to_end(report_path) 

92 while len(_CACHE) > MAX_CACHED_REPORTS: 92 ↛ 93line 92 didn't jump to line 93 because the condition on line 92 was never true

93 _CACHE.popitem(last=False) 

94 return _as_result(scores=scores)

97def _no_report() -> EvaluationReportScores: 

98 """The result for a run without a report, with a fresh list the caller may keep.""" 

99 return EvaluationReportScores(has_report=False, scores=[])

102def _as_result(scores: tuple[EvaluationScore, ...] | None) -> EvaluationReportScores: 

103 """The result for a report that exists, with a fresh list the caller may keep.""" 

104 if scores is None: 

105 return EvaluationReportScores(has_report=True, scores=None) 

106 return EvaluationReportScores(has_report=True, scores=list(scores))

108 

109def _parse_scores(report_path: Path) -> tuple[EvaluationScore, ...] | None: 

110 """Read the report's scores, or None when the file cannot be read or does not parse.""" 

111 try: 

112 report = _ReportScores.model_validate_json(report_path.read_bytes()) 

113 except OSError, ValidationError: 

114 logger.exception("Failed to read evaluation scores from %s", report_path) 

115 return None 

116 return tuple(report.measurements)
