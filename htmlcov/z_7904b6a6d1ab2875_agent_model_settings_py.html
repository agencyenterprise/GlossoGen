1"""The model settings a run gives its simulation agents and its LLM judges. 

2 

3``AgentModelSettings`` and ``JudgeModelSettings`` are the subset of pydantic-ai's 

4``ModelSettings`` a run may set, each field named and typed as pydantic-ai names and 

5types it. The agents' settings travel whole from the ``agent_model_settings`` knob, 

6or an agent's own ``model_overrides`` entry, to each agent's config, its 

7``AgentRegistered`` event, the runner and the probe metrics. A judge's come from a 

8scenario's ``judge_model_settings`` knob for its round judge, and from 

9``evaluate --judge-settings`` for the evaluation judge. Both extend 

10``ModelRequestSettings``, so adding a setting is adding a field there; they differ 

11only in the output cap, which an agent always sends and a judge sends only when it 

12is set. 

13 

14It is a model of its own rather than ``ModelSettings`` itself because a run's 

15settings are recorded in its log and shown on its page: they have to be plain 

16JSON and reject keys nobody defined. That also leaves out the settings that would 

17break a run or leak from it: ``tool_choice`` (the runner relies on the model 

18picking its own tools) and ``extra_headers`` (a header could carry a credential 

19into a log that gets exported). 

20""" 

21

30def _is_unset(value: object) -> bool: 

31 """Whether a setting is left out of the JSON: an unset one was not sent.""" 

32 return value is None

35class ModelRequestSettings(BaseModel): 

36 """The settings agents and judges share: every one but the output cap. 

37 

38 Every field defaults to ``None``, which sends nothing and leaves the 

39 provider's own default in place. An unset field is also left out of the 

40 JSON, so a log records only what was sent. pydantic-ai documents each setting 

41 and which providers honour it; a provider that does not accept a value fails 

42 the call. 

43 

44 ``thinking`` is how hard the model reasons: ``true``, ``false``, or one of 

45 ``minimal``, ``low``, ``medium``, ``high``, ``xhigh``, ``max``, translated by 

46 pydantic-ai for each provider. 

47 

48 ``timeout`` is in seconds. pydantic-ai also takes an httpx ``Timeout``, which 

49 has no JSON form. 

50 

51 ``extra_body`` is merged into the request body as given, for a provider 

52 parameter pydantic-ai does not model. Nothing checks it. 

53 """ 

54 

55 model_config = ConfigDict(extra="forbid") 

56 

57 thinking: ThinkingLevel | None = Field(default=None, exclude_if=_is_unset) 

58 temperature: float | None = Field(default=None, exclude_if=_is_unset) 

59 top_p: float | None = Field(default=None, exclude_if=_is_unset) 

60 top_k: int | None = Field(default=None, exclude_if=_is_unset) 

61 seed: int | None = Field(default=None, exclude_if=_is_unset) 

62 presence_penalty: float | None = Field(default=None, exclude_if=_is_unset) 

63 frequency_penalty: float | None = Field(default=None, exclude_if=_is_unset) 

64 logit_bias: dict[str, int] | None = Field(default=None, exclude_if=_is_unset) 

65 stop_sequences: list[str] | None = Field(default=None, exclude_if=_is_unset) 

66 service_tier: ServiceTier | None = Field(default=None, exclude_if=_is_unset) 

67 timeout: float | None = Field(default=None, gt=0, exclude_if=_is_unset) 

68 parallel_tool_calls: bool | None = Field(default=None, exclude_if=_is_unset) 

69 extra_body: dict[str, JsonValue] | None = Field(default=None, exclude_if=_is_unset)

71 def to_pydantic_ai(self) -> ModelSettings: 

72 """Return the settings as pydantic-ai's ``ModelSettings``, leaving unset ones out. 

73 

74 Every field shares its name with a ``ModelSettings`` key, so the dump is 

75 the mapping; an unset field is left out of it. 

76 """ 

77 return cast(ModelSettings, self.model_dump())

80class AgentModelSettings(ModelRequestSettings): 

81 """Settings sent with every request an agent makes. 

82 

83 ``max_tokens`` caps one response, reasoning tokens included, and is always 

84 sent. The default is sized for thinking-capable models; self-hosted 

85 deployments served at a small fixed context often need it lower, since input 

86 plus ``max_tokens`` has to fit. 

87 """ 

88 

89 max_tokens: int = Field(default=DEFAULT_AGENT_MAX_TOKENS, ge=1) 

90 

91 

92class JudgeModelSettings(ModelRequestSettings): 

93 """Settings sent with every call an LLM judge makes. 

94 

95 ``max_tokens`` is unset by default like every other field, so a judge's 

96 output is capped by the provider: on Anthropic, pydantic-ai sends the model's 

97 own maximum output; on OpenAI, the model's limit applies. 

98 """ 

99 

100 max_tokens: int | None = Field(default=None, ge=1, exclude_if=_is_unset)
