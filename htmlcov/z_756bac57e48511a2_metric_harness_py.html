1"""Score a finished simulation the way `glossogen evaluate` does. 

2 

3A metric's contract is not "given this list of events, return this number". It 

4is "given a run directory on disk, produce a report". Between the two sit the 

5JSONL reader, the round-transcript builder, the metric registry, the report 

6merge and the cost accounting, and a metric can be correct while any of those 

7mis-serves it. So this drives `run_scenario_evaluation`, the same function the 

8CLI calls, against a run that a real simulation actually wrote. 

9 

10Only the judge is replaced. Deterministic metrics never notice; LLM-judge 

11metrics get a `StubLLMProvider` whose answers the test chose, which is also 

12where the interesting assertion usually lives: not what the judge said, but 

13what it was shown. 

14"""

65def isolated_run(*, run: MetricRun, tmp_path: Path) -> MetricRun: 

66 """Copy `run`'s directory so a metric can write sidecars without leaking. 

67 

68 Several metrics persist files next to the log: `protocol_probe` appends 

69 `protocol_probe_responses.jsonl`, the similarity metrics and the 

70 communication pass each write their own JSON. The shared run is reused by 

71 every test in the package, so writing into it makes results depend on which 

72 test ran first, and an appending metric makes them depend on how many times. 

73 """ 

74 copied_dir = tmp_path / "run" 

75 shutil.copytree(run.run_dir, copied_dir) 

76 return MetricRun( 

77 scenario=run.scenario, 

78 run_dir=copied_dir, 

79 log_path=copied_dir / run.log_path.name, 

80 simulation=run.simulation, 

81 )

114# pydantic-ai routes a structured `output_type` through a tool call rather than 

115# free text, under this name. A probe answer therefore has to arrive as a call 

116# to it, not as a `SayTurn`. 

117STRUCTURED_OUTPUT_TOOL = "final_result"

120def use_scripted_probe_model( 

121 *, 

122 answers: list[str], 

123 output_type: type[BaseModel], 

124 monkeypatch: pytest.MonkeyPatch, 

125) -> None: 

126 """Make the probe metrics answer from a script instead of a provider. 

127 

128 `protocol_probe` and `protocol_explanation` do not go through the judge 

129 provider. They build their own pydantic-ai agent per probed agent, under 

130 that agent's original model, so `StubLLMProvider` never sees them and 

131 without this they would reach a real API on every run of the suite. 

132 

133 `output_type` is the schema that metric asks for, and it differs between 

134 them: the probe wants a reasoning/message pair, the explanation wants a 

135 single description. The args are built from its fields, because a fixed 

136 payload validates for one and not the other, and a probe whose answer fails 

137 validation is swallowed by the metric's own error handling. That looks like 

138 an agent with no history rather than a broken fake. 

139 

140 Each answer becomes one probe response, in order; the script repeats its 

141 last answer once spent, because the call count is the question bank times 

142 the replica count and a test should not have to restate that arithmetic. 

143 """ 

144 fields = list(output_type.model_fields) 

145 turns = [ 

146 ToolTurn( 

147 tool_name=STRUCTURED_OUTPUT_TOOL, 

148 # Every field is filled, and the last one carries the answer, so a 

149 # test can assert on the text whichever schema it asked for. 

150 args={name: (answer if name == fields[-1] else "scripted") for name in fields}, 

151 ) 

152 for answer in answers 

153 ] 

154 scripted = build_scripted_model(turns=turns, when_exhausted=[turns[-1]]) 

155 

156 def probe_model(model: str, provider: str) -> object: 

157 """Stand in for the per-agent model factory the probe agent builds.""" 

158 _ = model, provider 

159 return scripted 

160 

161 monkeypatch.setattr( 

162 "glossogen.evaluation.metrics.protocol_probe.probe_agent.build_pydantic_ai_model", 

163 probe_model, 

164 )

167async def score_metrics( 

168 *, 

169 run: MetricRun, 

170 metric_names: list[str], 

171 judge_responses: list[BaseModel], 

172 options: MetricRunOptions, 

173 report_path: Path, 

174 monkeypatch: pytest.MonkeyPatch, 

175) -> ScoredRun: 

176 """Run `metric_names` over `run` and return the report. 

177 

178 `judge_responses` are handed to the stub judge in order, one per 

179 `generate_structured` call. Pass an empty list for deterministic metrics: an 

180 unexpected call then fails with the schema it asked for, rather than a 

181 metric quietly scoring on an invented answer. 

182 

183 `report_path` is explicit because evaluation merges into whatever is already 

184 there. Tests that want a clean report pass a path of their own; a test about 

185 merging passes the same path twice. Defaulting it to somewhere inside the 

186 run directory would silently couple every test sharing that run to the order 

187 they happen to execute in. 

188 """ 

189 judge = StubLLMProvider() 

190 for response in judge_responses: 

191 judge.queue(response=response)

198 ) -> LLMProvider:

200 _ = provider_name, model, inference_provider, reasoning_effort 

201 return judge
