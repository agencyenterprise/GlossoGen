1"""Building the rewind state a resumed run starts from. 

2 

3One entry point, :func:`load_resume_state`, covers the resumed shapes: a 

4cross-run replace-agent run (an imported agent carries another run's history), a 

5replace-agent or fork-at-round run (anchored at a fork boundary by its 

6manifest), and a plain ``--resume`` of an interrupted run (no manifest at all). 

7 

8A fork whose boundary was the source's final round has no ``RoundAdvanced`` for 

9the round it should enter: its clone ends with the boundary round completed. 

10:func:`load_resume_state` detects that from the log itself (the last advanced 

11round is one behind the manifest's entry round) and marks the state with 

12``enter_round_by_advancing`` so the supervisor advances into the entry round 

13instead of re-opening a finished one. 

14 

15Crash recovery classifies what the log holds past the manifest's anchor. Agent 

16re-registrations alone mean a launch that never got going, so the boundary 

17anchor still applies. Agent activity is the event types history reconstruction 

18consumes, an exact set by construction; everything else (a fresh advance, 

19delivered injections, a round that ended with no agent activity, scenario and 

20world events the clock flushes while opening a round) is progress the anchor 

21would replay, so recovery re-anchors at the log's end: the state walk then 

22carries the already-logged advance, injections, and verdicts, and none are 

23recorded twice. Play is also recovered at the log's end for replace-agent and 

24fork-at-round runs, with the manifest's seeding filters bounded to the 

25predecessor's rounds so the replacement keeps its own turns. A cross-run fork 

26whose imported agent may have played is refused instead: its history is 

27rebuilt exclusively from source B's events, so post-boundary turns cannot be 

28re-seeded. 

29"""

126def _channel_visibility_from_manifest( 

127 visible_channels: list[str], 

128 blocked_channels: list[str], 

129 history_floors: dict[str, int], 

130) -> dict[str, ChannelVisibility]: 

131 """Translate replace-agent manifest channel lists into a visibility dict. 

132 

133 ``visible_channels`` (channels whose prior history remains visible) 

134 map to ``ChannelVisibilityFull``, except channels named in 

135 ``history_floors`` which map to ``ChannelVisibilityFromRound`` (their 

136 history is windowed from the floor round onward). ``blocked_channels`` 

137 (channels whose tool calls are stripped from the predecessor's 

138 history) map to ``ChannelVisibilityNone``. Channels not in any list 

139 are omitted (caller decides default behaviour). 

140 """ 

141 result: dict[str, ChannelVisibility] = {} 

142 for channel_id in visible_channels: 

143 floor = history_floors.get(channel_id) 

144 if floor is None: 

145 result[channel_id] = ChannelVisibilityFull() 

146 else: 

147 result[channel_id] = ChannelVisibilityFromRound(round_floor=floor) 

148 for channel_id in blocked_channels: 

149 result[channel_id] = ChannelVisibilityNone() 

150 return result

192def classify_fork_progress( 

193 events: list[SimulationEvent], 

194 target_event_id: str, 

195) -> ForkProgress: 

196 """Classify what the fork's log holds past its boundary anchor. 

197 

198 A launch appends agent re-registrations before anything happens, so those 

199 alone still mean a first launch (``PRISTINE``). Agent activity means the 

200 fork played (``PLAYED``); the set of event types agents produce is exact, 

201 because history reconstruction reads precisely those types. Anything else 

202 is clock, world, or scenario lifecycle the clock flushes while opening a 

203 round (a fresh advance, injections, a scenario's case events from 

204 ``on_round_advanced``), which a boundary-anchored rebuild would replay 

205 (``ADVANCED``). 

206 """ 

207 anchor_seen = False 

208 progress = ForkProgress.PRISTINE 

209 for event in events: 

210 if anchor_seen: 

211 if isinstance(event, _AGENT_ACTIVITY_EVENT_TYPES): 

212 return ForkProgress.PLAYED 

213 if isinstance(event, _LAUNCH_BOOKKEEPING_EVENT_TYPES): 

214 continue 

215 progress = ForkProgress.ADVANCED 

216 continue 

217 if event.event_id == target_event_id: 

218 anchor_seen = True 

219 return progress

222def apply_fork_boundary(state: RewindState, entry_round: int) -> RewindState: 

223 """Mark the state to advance when the clone ends at a completed boundary. 

224 

225 A fork whose boundary was the source's final round has no 

226 ``RoundAdvanced(entry_round)``: the clone's last advanced round is 

227 ``entry_round - 1``. The supervisor must then advance into the entry round. 

228 A state at or past the entry round needs no advance; recovery of a fork 

229 that already logged its fresh ``RoundAdvanced`` lands here, which is what 

230 keeps the advance from being recorded twice. 

231 The message-count snapshot for the entry round is synthesized from the 

232 walked messages, because ``ChannelVisibilityFromRound(entry_round)`` on a 

233 replaced agent would otherwise find no snapshot and fall back to a join 

234 index of zero, silently granting full history. 

235 """ 

236 if state.round_number >= entry_round: 

237 return state 

238 if state.round_number != entry_round - 1: 

239 raise ValueError( 

240 f"clone is inconsistent with its manifest: last advanced round is " 

241 f"{state.round_number} but the manifest enters round {entry_round}" 

242 ) 

243 snapshot = dict(state.channel_message_count_at_round_start) 

244 snapshot[entry_round] = { 

245 channel_id: len(messages) for channel_id, messages in state.messages_by_channel.items() 

246 } 

247 return state._replace( 

248 enter_round_by_advancing=True, 

249 channel_message_count_at_round_start=snapshot, 

250 )

265def _boundary_anchored_visibility( 

266 channel_visibility: dict[str, ChannelVisibility], 

267 entry_round: int, 

268) -> dict[str, ChannelVisibility]: 

269 """Anchor hidden channels at the fork boundary instead of the recovery point. 

270 

271 ``ChannelVisibilityNone`` translates to a join index of every message so 

272 far, which at recovery time is the crash point: messages the replacement 

273 had already seen on that channel would become hidden. ``FromRound`` at the 

274 entry round pins the join index to the boundary count the original launch 

275 used, read from the entry round's message-count snapshot. 

276 """ 

277 result: dict[str, ChannelVisibility] = {} 

278 for channel_id, visibility in channel_visibility.items(): 

279 if isinstance(visibility, ChannelVisibilityNone): 

280 result[channel_id] = ChannelVisibilityFromRound(round_floor=entry_round) 

281 else: 

282 result[channel_id] = visibility 

283 return result

432

434 crash recovery strips only the predecessor's turns: the replacement's own 

435 post-boundary text, thinking, and channel traffic stay in its history.

438 events=events, 

439 target_event_id=replace_info.target_event_id, 

440 )

442 replaced_agent_id: AgentHistoryFilter( 

443 tool_calls_only=True, 

444 channel_visibility=replace_info.channel_visibility, 

445 imported=None, 

446 filter_below_round=replace_info.entry_round, 

447 split_parallel_tool_calls=replace_info.replacement_provider == SELF_HOSTED_PROVIDER, 

448 ) 

449 }

451 cutoff_round: int | None = replace_info.entry_round

453 logger.info(

457 )

463 # had already seen. The history filter keeps the manifest's original

474 events=events, 

475 target_event_id=_resume_anchor_event_id( 

476 events=events, 

477 target_event_id=replace_info.target_event_id, 

478 progress=progress, 

479 ),
