1"""Drive 50 baseline veyru sonnet-4.6 runs (10 configs x 5 reps) at concurrency=3. 

2 

3After each sim finishes successfully, writes labels.json with the standard 

4baseline tags plus `no_specialist`, so the new batch is distinguishable from 

5the older baseline batch. 

6""" 

7 

8import asyncio 

9import json 

10import re 

11import sys 

12import time 

13from pathlib import Path 

14from typing import NamedTuple 

15 

16 

17class Job(NamedTuple): 

18 config: str 

19 budget: int 

20 pm: str 

21 pm_label: str 

22 rep: int 

23 

24 

25CONCURRENCY = 3 

26RUNS_DIR = Path("runs") 

27DRIVER_LOG = RUNS_DIR / "baseline_no_specialist_driver.log" 

28STDOUT_LOG_DIR = RUNS_DIR / "_no_specialist_stdouts" 

29STDOUT_LOG_DIR.mkdir(parents=True, exist_ok=True) 

30 

31LABELS_BASE = ["baseline", "no_specialist", "sonnet-4.6", "single_team"] 

32BUDGETS = [150, 250, 450, 800, 2000] 

33PM_VARIANTS = [ 

34 ("on", "postmortem=true", "knobs_baseline_budget_{budget}.json"), 

35 ("off", "postmortem=false", "knobs_baseline_no_postmortem_budget_{budget}.json"), 

36] 

37REPS = 5 

38 

39 

40def build_queue() -> list[Job]: 

41 queue: list[Job] = [] 

42 for budget in BUDGETS: 

43 for pm_state, pm_label, cfg_template in PM_VARIANTS: 

44 cfg_file = f"src/glossogen/scenarios/veyru/{cfg_template.format(budget=budget)}" 

45 for rep in range(REPS): 

46 queue.append( 

47 Job( 

48 config=cfg_file, 

49 budget=budget, 

50 pm=pm_state, 

51 pm_label=pm_label, 

52 rep=rep, 

53 ) 

54 ) 

55 return queue 

56 

57 

58def log(msg: str) -> None: 

59 line = f"[{time.strftime('%Y-%m-%d %H:%M:%S')}] {msg}" 

60 print(line, flush=True) 

61 with DRIVER_LOG.open("a") as f: 

62 f.write(line + "\n") 

63 

64 

65RUN_DIR_RE = re.compile(r"Run directory:\s+(runs/veyru/[0-9_]+)") 

66 

67 

68def find_run_dir(stdout_log: Path) -> Path | None: 

69 if not stdout_log.exists(): 

70 return None 

71 text = stdout_log.read_text(errors="replace") 

72 match = RUN_DIR_RE.search(text) 

73 if not match: 

74 return None 

75 return Path(match.group(1)) 

76 

77 

78async def launch_job(idx: int, job: Job) -> None: 

79 stdout_log = STDOUT_LOG_DIR / f"job_{idx:03d}.log" 

80 cmd = ( 

81 "VIRTUAL_ENV= uv run --no-sync python -m glossogen run veyru " 

82 "--model claude-sonnet-4-6 --provider anthropic --runs-dir ./runs " 

83 f"--config {job.config}" 

84 ) 

85 log(f"START idx={idx:03d} budget={job.budget} pm={job.pm} " f"rep={job.rep} -> {stdout_log}") 

86 with stdout_log.open("w") as f: 

87 proc = await asyncio.create_subprocess_shell( 

88 cmd, stdout=f, stderr=asyncio.subprocess.STDOUT 

89 ) 

90 rc = await proc.wait() 

91 log(f"EXIT idx={idx:03d} rc={rc}") 

92 if rc != 0: 

93 log(f"FAIL idx={idx:03d} non-zero exit; not labelling") 

94 return 

95 run_dir = find_run_dir(stdout_log) 

96 if run_dir is None: 

97 log(f"WARN idx={idx:03d} could not parse run dir from stdout") 

98 return 

99 labels = sorted(LABELS_BASE + [f"budget={job.budget}", job.pm_label]) 

100 labels_path = run_dir / "labels.json" 

101 labels_path.write_text(json.dumps(labels)) 

102 log(f"LABEL idx={idx:03d} {run_dir.name} {labels}") 

103 

104 

105async def worker(worker_id: int, queue: asyncio.Queue[tuple[int, Job]]) -> None: 

106 while True: 

107 try: 

108 idx, job = queue.get_nowait() 

109 except asyncio.QueueEmpty: 

110 log(f"worker {worker_id} drained queue, exiting") 

111 return 

112 try: 

113 await launch_job(idx, job) 

114 finally: 

115 queue.task_done() 

116 

117 

118async def main() -> None: 

119 jobs = build_queue() 

120 log(f"== driver start: {len(jobs)} jobs, concurrency={CONCURRENCY} ==") 

121 queue: asyncio.Queue[tuple[int, Job]] = asyncio.Queue() 

122 for i, job in enumerate(jobs): 

123 queue.put_nowait((i, job)) 

124 workers = [asyncio.create_task(worker(i, queue)) for i in range(CONCURRENCY)] 

125 await asyncio.gather(*workers) 

126 log("== driver done ==") 

127 

128 

129if __name__ == "__main__": 

130 sys.exit(asyncio.run(main()))
