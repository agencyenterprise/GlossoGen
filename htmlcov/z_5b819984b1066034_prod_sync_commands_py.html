163async def handle_login(args: argparse.Namespace) -> None: 

164 """Drive the ``glossogen login`` subcommand.""" 

165 try: 

166 credentials = await run_login( 

167 issuer_url=args.url, 

168 timeout_seconds=args.timeout_seconds, 

169 ) 

170 except Exception as exc: 

171 logger.exception("Login failed") 

172 raise SystemExit(f"Login failed: {exc}") from exc 

173 print( 

174 f"Logged in to {credentials.issuer_url} as group " 

175 f"{credentials.group_slug!r}. " 

176 f"Credentials saved to {CREDENTIALS_PATH}." 

177 )

180async def handle_push_to_prod(args: argparse.Namespace) -> None: 

181 """Drive the ``glossogen push-to-prod`` subcommand.""" 

182 concurrency = max(1, min(int(args.concurrency), 16)) 

183 scenarios_arg: list[str] = args.scenarios 

184 scenarios = frozenset(scenarios_arg) if scenarios_arg else None 

185 spec = PushSpec( 

186 runs_dir=Path(args.runs_dir), 

187 labels=frozenset(args.labels), 

188 scenarios=scenarios, 

189 require_report=not args.include_incomplete, 

190 dry_run=args.dry_run, 

191 concurrency=concurrency, 

192 ) 

193 tally = await run_push_to_prod(spec=spec) 

194 print( 

195 f"Done. uploaded={len(tally.uploaded)} " 

196 f"skipped={len(tally.skipped)} " 

197 f"failed={len(tally.failed)}" 

198 ) 

199 if tally.failed: 

200 raise SystemExit(1)
