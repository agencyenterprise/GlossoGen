1"""Filtering runs by the values in their recorded ``scenario_config``. 

2 

3A filter is written as one string, ``<knob><operator><value>``, so a whole set 

4of them travels as repeated query parameters and as a JSON array of strings in 

5an export body, without either shape needing a nested object. 

6``time_budget_seconds>=200`` and ``postmortem_enabled=true`` are both filters. 

7 

8A nested knob is addressed with dots, the way the CSV export names its column: 

9``model_overrides.field_observer.model=gpt-5.4``. 

10 

11A knob a run recorded as null is filterable. ``swap_round=null`` selects the runs 

12that never swapped and ``swap_round!=null`` the ones that did, so the two 

13partition the runs recording that knob. ``swap_round!=16`` includes a run that 

14never swapped, because not swapping at all is not swapping at 16. A knob the run 

15never recorded at all is different, and matches nothing. 

16 

17The knob name ends at the **first** operator in the string, and the longest 

18operator starting there wins. A value may therefore contain an operator 

19character, which a model name does: ``judge_model=gpt>=5`` filters ``judge_model`` 

20for the value ``gpt>=5``. 

21 

22It sits at the package root because three layers share it: the listing that 

23answers the runs page, the export request model that validates a selection, and 

24the resolver the CLI reaches with no server in front. Holding it under any one 

25of those would make the other two import across a boundary their own docstrings 

26disclaim. 

27 

28Comparison is typed from the run's own recorded value rather than from the 

29scenario's knobs schema. The listing already holds the config it is filtering, 

30and reading the type off it means this module never imports a scenario class, 

31which keeps the schema out of the request path and lets a run recorded against 

32an older schema still answer. 

33""" 

34 

35import logging 

36from enum import Enum 

37from typing import Any, NamedTuple, cast

39logger = logging.getLogger(__name__)

42class KnobFilterOperator(str, Enum): 

43 """The comparisons a knob filter can express.""" 

44 

45 EQ = "=" 

46 NE = "!=" 

47 GE = ">=" 

48 LE = "<=" 

49 GT = ">" 

50 LT = "<" 

51 

52 

53# Ordered so that, among the operators starting at the same index, the longest 

54# is seen first: ">=" is never read as ">" followed by a value of "=200". 

55_OPERATORS_BY_LENGTH = sorted(KnobFilterOperator, key=lambda op: -len(op.value)) 

56 

57_TRUE_TEXT = frozenset({"true", "1", "yes", "on"}) 

58_FALSE_TEXT = frozenset({"false", "0", "no", "off"}) 

59 

60# What a filter writes to mean "this knob is not set". One spelling, because 

61# each one shadows a literal value a string or enum knob could hold: with 

62# "none" reserved, `mode=none` could never ask about a mode called none. The 

63# empty value is not among them either, since `judge_model=` asks for the empty 

64# string, which for a string knob is a different question from being unset. 

65_NULL_VALUE_TEXT = "null" 

66 

67_ORDERING_OPERATORS = frozenset( 

68 { 

69 KnobFilterOperator.GE, 

70 KnobFilterOperator.LE, 

71 KnobFilterOperator.GT, 

72 KnobFilterOperator.LT, 

73 } 

74)

77class KnobFilter(NamedTuple): 

78 """One parsed ``<knob><operator><value>`` condition. 

79 

80 ``value_text`` stays as written. It is coerced per run, against the type of 

81 that run's own recorded value. 

82 """ 

83 

84 knob: str 

85 operator: KnobFilterOperator 

86 value_text: str 

87 

88 

89class KnobFilterParseError(ValueError): 

90 """Raised for a filter string carrying no operator, or an empty knob name.""" 

91 

92 

93def _find_separator(raw: str) -> tuple[int, KnobFilterOperator] | None: 

94 """Locate the operator that separates the knob name from the value. 

95 

96 The earliest operator wins, and among those starting at that index the 

97 longest one does. Scanning for the longest operator anywhere instead would 

98 let an operator character inside the value take over: ``judge_model=gpt>=5`` 

99 would split on ``>=`` and ask about a knob named ``judge_model=gpt``. 

100 

101 An operator at index 0 is still the separator. Skipping it and taking the 

102 next one would read ``>=200`` as a knob named ``>`` compared against 

103 ``200``, which parses cleanly and then matches nothing. 

104 """ 

105 best: tuple[int, KnobFilterOperator] | None = None 

106 for operator in _OPERATORS_BY_LENGTH: 

107 index = raw.find(operator.value) 

108 if index < 0: 

109 continue 

110 if best is None or index < best[0]: 

111 best = (index, operator) 

112 return best

115def parse_knob_filter(raw: str) -> KnobFilter: 

116 """Parse one ``<knob><operator><value>`` string. 

117 

118 Raises :class:`KnobFilterParseError` when no operator appears, or when the 

119 knob name is empty. An empty value is allowed: it matches a knob whose 

120 recorded value is the empty string. 

121 """ 

122 found = _find_separator(raw=raw) 

123 if found is None: 

124 raise KnobFilterParseError( 

125 f"Knob filter {raw!r} carries no operator. Expected <knob><operator><value>, " 

126 f"with the operator one of {', '.join(op.value for op in KnobFilterOperator)}." 

127 ) 

128 index, operator = found 

129 knob = raw[:index].strip() 

130 if not knob: 

131 raise KnobFilterParseError( 

132 f"Knob filter {raw!r} names no knob before its {operator.value!r} operator. " 

133 f"Expected <knob><operator><value>." 

134 ) 

135 return KnobFilter( 

136 knob=knob, 

137 operator=operator, 

138 value_text=raw[index + len(operator.value) :].strip(), 

139 ) 

140 

141 

142def parse_knob_filters(raw_filters: list[str]) -> list[KnobFilter]: 

143 """Parse every filter string, preserving order.""" 

144 return [parse_knob_filter(raw=raw) for raw in raw_filters]

147def knob_filter_problem(raw_filters: list[str]) -> str | None: 

148 """Why these strings are not all conditions, or None when they are. 

149 

150 The non-raising half of the pair, for a caller that reports the problem 

151 itself. The CLI is one: catching the exception there would oblige it to log a 

152 stack trace for what is a mistyped flag. 

153 """ 

154 for raw in raw_filters: 

155 try: 

156 parse_knob_filter(raw=raw) 

157 except KnobFilterParseError as exc: 

158 return str(exc) 

159 return None

162def _coerce_to_bool(text: str) -> bool | None: 

163 """Read a boolean from a filter's value text, or None when it names neither.""" 

164 lowered = text.strip().lower() 

165 if lowered in _TRUE_TEXT: 

166 return True 

167 if lowered in _FALSE_TEXT: 

168 return False 

169 return None

172def _coerce_to_float(text: str) -> float | None: 

173 """Read a number from a filter's value text, or None when it is not one.""" 

174 try: 

175 return float(text) 

176 except ValueError: 

177 return None

180def _compare(left: Any, right: Any, operator: KnobFilterOperator) -> bool: 

181 """Apply an operator to two values already coerced to a comparable pair.""" 

182 if operator is KnobFilterOperator.EQ: 

183 return bool(left == right) 

184 if operator is KnobFilterOperator.NE: 

185 return bool(left != right) 

186 if operator is KnobFilterOperator.GE: 

187 return bool(left >= right) 

188 if operator is KnobFilterOperator.LE: 

189 return bool(left <= right) 

190 if operator is KnobFilterOperator.GT: 

191 return bool(left > right) 

192 return bool(left < right) 

193

195def _matches_bool(recorded: bool, knob_filter: KnobFilter) -> bool: 

196 """Compare a boolean knob. Ordering operators do not apply to one.""" 

197 if knob_filter.operator in _ORDERING_OPERATORS: 

198 return False 

199 wanted = _coerce_to_bool(text=knob_filter.value_text) 

200 if wanted is None: 

201 return False 

202 return _compare(left=recorded, right=wanted, operator=knob_filter.operator) 

203 

204 

205def _matches_number(recorded: float, knob_filter: KnobFilter) -> bool: 

206 """Compare a numeric knob.""" 

207 wanted = _coerce_to_float(text=knob_filter.value_text) 

208 if wanted is None: 

209 return False 

210 return _compare(left=recorded, right=wanted, operator=knob_filter.operator)

213def _names_null(value_text: str) -> bool: 

214 """Whether a condition's value names the unset state rather than a value.""" 

215 return value_text.strip().casefold() == _NULL_VALUE_TEXT

218def _matches_set_question(recorded: Any, knob_filter: KnobFilter) -> bool: 

219 """Answer a condition whose value names null, which asks only about set-ness. 

220 

221 Handled before the type dispatch, because the question is about the absence 

222 of a value rather than about one: a run recording ``16`` has to answer 

223 ``swap_round!=null`` with yes, and the numeric comparison cannot, since 

224 "null" is not a number. 

225 """ 

226 if knob_filter.operator in _ORDERING_OPERATORS: 

227 return False 

228 if knob_filter.operator is KnobFilterOperator.EQ: 

229 return recorded is None 

230 return recorded is not None 

231 

232 

233def _matches_recorded_null(knob_filter: KnobFilter) -> bool: 

234 """Compare a null-recorded knob against a condition naming a concrete value. 

235 

236 A recorded null is an answer, not the absence of one: an optional knob left 

237 unset says the run did not swap, or ran with no budget cap. So it is not 

238 equal to any concrete value, it is unequal to every one of them (a run that 

239 never swapped did not swap at round 16), and no ordering holds, since null 

240 sits nowhere on the number line. 

241 """ 

242 if knob_filter.operator in _ORDERING_OPERATORS: 

243 return False 

244 return knob_filter.operator is KnobFilterOperator.NE 

245

247def _matches_text(recorded: str, knob_filter: KnobFilter) -> bool: 

248 """Compare a knob as text, case-insensitively. 

249 

250 Ordering operators are refused rather than compared lexicographically: 

251 ``judge_model >= "claude"`` reads like a mistake, and answering it would 

252 hide one. 

253 """ 

254 if knob_filter.operator in _ORDERING_OPERATORS: 

255 return False 

256 return _compare( 

257 left=recorded.casefold(), 

258 right=knob_filter.value_text.casefold(), 

259 operator=knob_filter.operator, 

260 )

266def _lookup(scenario_config: dict[str, Any], knob: str) -> Any: 

267 """Read a knob's recorded value, following dots into nested mappings. 

268 

269 ``model_overrides.field_observer.model`` is how the CSV export names that 

270 column and how an analysis dimension addresses it, so a condition has to 

271 read it the same way. A top-level key wins over the dotted reading, since a 

272 scenario is free to declare a knob whose name contains a dot. 

273 

274 Returns :data:`_MISSING` when nothing is there, which is not the same as a 

275 recorded ``None``. 

276 """ 

277 if knob in scenario_config: 

278 return scenario_config[knob] 

279 if "." not in knob: 

280 return _MISSING 

281 current: Any = scenario_config 

282 for part in knob.split("."): 

283 if not isinstance(current, dict): 

284 return _MISSING 

285 step = cast(dict[str, Any], current) 

286 if part not in step: 

287 return _MISSING 

288 current = step[part] 

289 return current

292def _matches_one(scenario_config: dict[str, Any], knob_filter: KnobFilter) -> bool: 

293 """Test one filter against a run's recorded config. 

294 

295 A knob the run never recorded does not match, including under ``!=``: the 

296 run cannot answer the question, which is not the same as answering it in the 

297 negative. A knob recorded *as* null did answer, and is handled by 

298 :func:`_matches_recorded_null`. 

299 """ 

300 recorded = _lookup(scenario_config=scenario_config, knob=knob_filter.knob) 

301 if recorded is _MISSING: 

302 return False 

303 if _names_null(value_text=knob_filter.value_text): 

304 return _matches_set_question(recorded=recorded, knob_filter=knob_filter) 

305 if recorded is None: 

306 return _matches_recorded_null(knob_filter=knob_filter) 

307 # bool before the numeric branch: in Python a bool is an int, and comparing 

308 # True against 1 would make postmortem_enabled>=1 a boolean ordering test. 

309 if isinstance(recorded, bool): 

310 return _matches_bool(recorded=recorded, knob_filter=knob_filter) 

311 if isinstance(recorded, (int, float)): 

312 return _matches_number(recorded=float(recorded), knob_filter=knob_filter) 

313 if isinstance(recorded, str): 

314 return _matches_text(recorded=recorded, knob_filter=knob_filter) 

315 # A list or a mapping has no scalar comparison. The schema endpoint does not 

316 # offer one, so reaching here means a hand-written filter. 

317 return False

320def matches_knob_filters( 

321 scenario_config: dict[str, Any], 

322 knob_filters: list[KnobFilter], 

323) -> bool: 

324 """Whether a run's config satisfies every filter (AND semantics).""" 

325 return all( 

326 _matches_one(scenario_config=scenario_config, knob_filter=knob_filter) 

327 for knob_filter in knob_filters 

328 )
