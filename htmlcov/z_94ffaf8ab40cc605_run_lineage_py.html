1"""Recovers a derived run's timeline parent from its on-disk manifests. 

2 

3A run created via ``replace-agent``, ``resume-at-round``, or 

4``cross-run-replace-agent`` records its timeline parent in a manifest file 

5(``replace_manifest.json`` for the first two, ``cross_run_replace_manifest.json`` 

6for the third, using source A as the timeline parent). This module reads those 

7manifests back into the ``(scenario, run_dir_name)`` identity the runs index 

8stores in its ``source_run_scenario`` / ``source_run_dir_name`` columns, so the 

9import flow can repopulate lineage that would otherwise be lost. 

10""" 

11 

12from pathlib import Path 

13from typing import NamedTuple 

14 

15from glossogen.cross_run_replace_manifest import read_cross_run_replace_manifest 

16from glossogen.replace_manifest import read_replace_manifest 

17 

18 

19class TimelineParent(NamedTuple): 

20 """A derived run's timeline-parent identity.""" 

21 

22 scenario: str 

23 run_dir_name: str 

24 

25 

26def read_timeline_parent(run_dir: Path) -> TimelineParent | None: 

27 """Return the timeline parent recorded in ``run_dir``'s manifests, else ``None``. 

28 

29 Cross-run derivations use source A (the timeline parent), matching the 

30 convention in the runs-index registration path. Returns ``None`` when the 

31 run carries neither a replace nor a cross-run manifest. 

32 """ 

33 cross_run_manifest = read_cross_run_replace_manifest(run_dir=run_dir) 

34 if cross_run_manifest is not None: 

35 return _split_run_id(run_id=cross_run_manifest.source_a_run_id) 

36 

37 replace_manifest = read_replace_manifest(run_dir=run_dir) 

38 if replace_manifest is not None: 

39 return _split_run_id(run_id=replace_manifest.source_run_id) 

40 

41 return None 

42 

43 

44def _split_run_id(run_id: str) -> TimelineParent: 

45 """Split a ``scenario/run_dir_name`` run id into its two components.""" 

46 scenario, run_dir_name = run_id.split("/", 1) 

47 return TimelineParent(scenario=scenario, run_dir_name=run_dir_name)
