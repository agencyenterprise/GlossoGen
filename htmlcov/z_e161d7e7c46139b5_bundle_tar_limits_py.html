1"""Bounds on how much reading an uploaded run bundle's tar headers may cost. 

2 

3``tarfile`` reads some headers whole into memory before it hands back the member they 

4describe: a GNU long name, a pax extended header, a GNU sparse map. Each one is sized 

5by the archive, so an upload of a few megabytes that gzip expands into gigabytes of 

6one header is read in full before any check on the members runs. Measured: a 1 MB 

7bundle holding one 1 GiB long-name header took the reading process to 7.7 GB. 

8 

9:class:`BundleTarFile` refuses those headers past a budget before reading them, and 

10refuses the header types a bundle never carries. 

11""" 

12 

13import tarfile 

14from collections.abc import Callable 

15from typing import cast 

16 

17# Every member header read is held in memory as a TarInfo, and an empty member costs 

18# the uploader almost nothing once gzipped. A run bundle carries a run directory's 

19# files, and a legacy one also its ``.git`` objects; this sits well above both. 

20MAX_BUNDLE_MEMBER_COUNT: int = 50_000 

21 

22# Long names and pax records across the whole archive. Python's default tar format 

23# writes a pax record of a few dozen bytes for each member with a fractional mtime, so 

24# this covers a bundle at the member ceiling. 

25MAX_BUNDLE_EXTENDED_HEADER_BYTES: int = 4 * 1024 * 1024 

26 

27# Extended headers in a row before one member. ``tarfile`` recurses once for each, so 

28# a long chain of empty ones ends in a RecursionError. A writer puts at most a pax 

29# header, or a long name and a long link, before a member. 

30MAX_CONSECUTIVE_EXTENDED_HEADERS: int = 4 

31 

32_EXTENDED_HEADER_TYPES: frozenset[bytes] = frozenset( 

33 { 

34 tarfile.GNUTYPE_LONGNAME, 

35 tarfile.GNUTYPE_LONGLINK, 

36 tarfile.XHDTYPE, 

37 tarfile.SOLARIS_XHDTYPE, 

38 } 

39) 

40 

41# A global pax header is copied into every later member, and a sparse member's map 

42# is read block by block with no bound. ``build_bundle_bytes`` writes neither. 

43_REFUSED_HEADER_TYPES: frozenset[bytes] = frozenset({tarfile.XGLTYPE, tarfile.GNUTYPE_SPARSE})

83class BundleTarFile(tarfile.TarFile):

93 def charge_extended_header(self, size: int) -> None: 

94 """Count one extended header of ``size`` bytes, refusing it past either bound.""" 

95 self.consecutive_extended_headers += 1 

96 if self.consecutive_extended_headers > MAX_CONSECUTIVE_EXTENDED_HEADERS:
