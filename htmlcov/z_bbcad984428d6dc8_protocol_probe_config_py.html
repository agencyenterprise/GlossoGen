1"""Configuration a scenario exposes to opt into the protocol-probe metric family. 

2 

3A scenario implements ``SimulationScenario.get_protocol_probe_config`` to 

4return one of these (or ``None`` to opt out). The platform's protocol-probe 

5metrics read the returned config rather than hard-coding scenario-specific 

6paths and role-name mappings. 

7 

8The test bank itself is read here too, so a caller that only needs to know 

9what the bank asks does not import the metric. 

10 

11``questions_path`` is a JSON file with the test bank; ``prompts_dir`` holds 

12the Jinja2 templates used to render each role's probe prompt; 

13``role_groups`` maps a question's ``agent_role_filter`` string to the set 

14of role names that should be probed for that filter (e.g. ``"observer"`` 

15expanding to ``{"Field Observer", "Field Observer A", "Field Observer B"}`` 

16in two-team scenarios); ``role_templates`` maps each filter to its template 

17file name inside ``prompts_dir``. 

18""" 

19 

20import json 

21from pathlib import Path 

22from typing import NamedTuple 

23 

24from pydantic import BaseModel 

25 

26 

27class ProtocolProbeConfig(NamedTuple): 

28 """Scenario-supplied configuration for the protocol-probe metric family.""" 

29 

30 questions_path: Path 

31 prompts_dir: Path 

32 role_groups: dict[str, frozenset[str]] 

33 role_templates: dict[str, str] 

34 

35 

36class ProbeQuestion(BaseModel): 

37 """One entry in the frozen test bank file. 

38 

39 ``inputs`` is rendered into the role's template, so its keys are the 

40 variables that template reads. A value keeps the JSON type it was written 

41 as: a template comparing one with ``==`` reads a quoted number as a string 

42 and takes the wrong branch. 

43 """ 

44 

45 id: str 

46 agent_role_filter: str 

47 inputs: dict[str, str | int] 

48 

49 

50def load_probe_questions(path: Path) -> list[ProbeQuestion]: 

51 """Read the frozen test bank JSON and validate each entry.""" 

52 raw = json.loads(path.read_text(encoding="utf-8")) 

53 return [ProbeQuestion.model_validate(entry) for entry in raw]
