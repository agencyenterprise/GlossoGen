1"""Apply in-run agent swaps on top of the configs rebuilt from registrations. 

2 

3``extract_agent_configs`` reads ``AgentRegistered`` events, which the supervisor 

4writes once per agent at startup. An in-run swap does not write another one: it 

5records ``AgentSwappedMidRun`` and updates the runtime in place. So a run whose 

6agent was swapped rebuilds with the *predecessor's* model and system prompt, 

7and anything that re-runs an agent from its config afterwards runs the wrong 

8agent. 

9 

10That matters for the probe metrics, which exist to ask an agent what it 

11remembers. Probing a swapped-in agent under its predecessor's model answers a 

12question nobody asked, and on a cross-model swap the answer comes from a 

13different provider entirely. 

14""" 

15 

16from glossogen.models.agent_config import AgentConfig 

17from glossogen.models.event import AgentSwappedMidRun, SimulationEvent 

18 

19 

20def apply_mid_run_swaps( 

21 agent_configs: list[AgentConfig], 

22 events: list[SimulationEvent], 

23) -> list[AgentConfig]: 

24 """Return the configs with each swapped agent moved onto its final model. 

25 

26 The last swap for an agent wins, so a multi-swap run resolves to the 

27 generation that was running when the simulation ended. Agents that were 

28 never swapped are returned unchanged. 

29 

30 The system prompt is left alone. A swap may carry its own 

31 (``SwapAgent.system_prompt``), but that is not recorded on the event, so 

32 inferring it here would be guesswork; the registration's prompt is the only 

33 one the log actually carries. 

34 """ 

35 latest_model: dict[str, AgentSwappedMidRun] = {} 

36 for event in events: 

37 if isinstance(event, AgentSwappedMidRun): 

38 latest_model[event.agent_id] = event 

39 

40 updated: list[AgentConfig] = [] 

41 for config in agent_configs: 

42 swap = latest_model.get(config.agent_id) 

43 if swap is None: 

44 updated.append(config) 

45 continue 

46 updated.append( 

47 config.model_copy(update={"model": swap.new_model, "provider": swap.new_provider}) 

48 ) 

49 return updated
