1"""Turning a selection into an ordered list of runs. 

2 

3Both surfaces reduce to filtering a `list[RunSummary]`, so the filtering itself 

4lives here once. The server path gets that list from the group-scoped listing; the 

5CLI path gets it from a filesystem walk. Nothing downstream of this module knows 

6which one it got. 

7 

8Runs come out sorted by id, not newest-first. Run ids are unique, so that is 

9a total order and the same selection puts its rows in the same order every time, 

10which makes the CSV bytes identical across runs. The archives are not byte-identical 

11even so: a zip stamps each member with the time it was written. 

12 

13The list's own newest-first display order is not a total order: two runs claimed in 

14the same second tie, and their relative position is then whatever the filesystem 

15returned. 

16"""

72def _sorted_by_run_id(summaries: list[RunSummary]) -> list[RunSummary]: 

73 """Return the runs in ascending run-id order.""" 

74 return sorted(summaries, key=lambda summary: summary.run_id)

77def partition_explicit_run_ids( 

78 run_ids: list[str], 

79 owned_by_run_id: dict[str, RunSummary], 

80) -> ResolvedSelection: 

81 """Split requested ids into runs that resolve and ids that do not. 

82 

83 Duplicates collapse. An id naming a run this caller cannot see comes back in 

84 ``missing_run_ids`` for the surface above to act on, since a row dropped here would 

85 leave no trace in the export. 

86 """ 

87 seen: set[str] = set() 

88 found: list[RunSummary] = [] 

89 missing: list[str] = [] 

90 for run_id in run_ids: 

91 if run_id in seen: 

92 continue 

93 seen.add(run_id) 

94 summary = owned_by_run_id.get(run_id) 

95 if summary is None: 

96 missing.append(run_id) 

97 continue 

98 found.append(summary) 

99 return ResolvedSelection(summaries=_sorted_by_run_id(found), missing_run_ids=missing)

108 return partition_explicit_run_ids(

110 owned_by_run_id={summary.run_id: summary for summary in candidates},

112 knob_filters = selection.parsed_knob_conditions() 

113 matched = [ 

114 summary 

115 for summary in candidates 

116 if _matches_filters(summary=summary, selection=selection, knob_filters=knob_filters) 

117 ] 

118 return ResolvedSelection(summaries=_sorted_by_run_id(matched), missing_run_ids=[])
