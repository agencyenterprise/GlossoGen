1"""Which scenario sources a process loads, resolved once at startup. 

2 

3The built-in package always comes first. After it, the first of these that is 

4set decides the rest: 

5 

61. ``--scenarios <source>`` on the command line, repeatable; 

72. ``GLOSSOGEN_SCENARIOS``, ``os.pathsep``-separated; 

83. ``[tool.glossogen] scenarios = [...]`` in the nearest ``pyproject.toml`` at or 

9 above the working directory, with paths relative to that file. 

10 

11When none is set, the project directory is scanned: the directory holding that 

12``pyproject.toml``, or the working directory when there is none. An empty list in 

13any of the three, ``scenarios = []`` and ``--scenarios ""`` included, means no 

14scan. 

15 

16A source is a path when it names an existing file or directory, and a module 

17name otherwise. Nothing here reads a run's data: what a process imports comes 

18only from its own command line, environment and project file. 

19 

20A child ``glossogen`` process gets ``GLOSSOGEN_SCENARIOS`` set to the modules its 

21parent loaded, and ``PYTHONPATH`` extended with the directories they import from 

22(:func:`child_process_environment`), so it imports the same modules under the 

23same names whatever its working directory, without scanning. 

24"""

76def find_project(working_dir: Path) -> ProjectLocation: 

77 """Return the project ``working_dir`` belongs to, walking up to the nearest pyproject.""" 

78 resolved = working_dir.resolve() 

79 for directory in (resolved, *resolved.parents): 

80 candidate = directory / PYPROJECT_NAME 

81 if candidate.is_file(): 

82 return ProjectLocation(root=directory, pyproject=candidate) 

83 return ProjectLocation(root=resolved, pyproject=None)

86def resolve_scenario_sources( 

87 cli_sources: Sequence[str], 

88 environ: Mapping[str, str], 

89 working_dir: Path, 

90 runs_dirs: Sequence[Path], 

91) -> tuple[ScenarioSource, ...]: 

92 """Return the sources this process loads: the built-ins, then the configured ones. 

93 

94 ``runs_dirs`` are the runs directories the command reads or writes; every 

95 directory source skips them, read relative to the working directory and to 

96 the project root, as it does the installed glossogen package. Blank 

97 ``--scenarios`` values are ignored, so ``--scenarios ""`` loads the built-ins 

98 only. 

99 Raises :class:`ScenarioCatalogError` for a source that is neither an existing 

100 path nor a module name, or a ``[tool.glossogen]`` table that cannot be read. 

101 """ 

102 project = find_project(working_dir=working_dir) 

103 skipped = _skipped_paths( 

104 working_dir=working_dir, 

105 project_root=project.root, 

106 environ=environ, 

107 runs_dirs=runs_dirs, 

108 ) 

109 builder = _SourceBuilder(project_root=project.root, skipped_paths=skipped) 

110 if cli_sources: 

111 configured = [ 

112 builder.source_for(spec=spec, base_dir=working_dir, origin="--scenarios") 

113 for spec in cli_sources 

114 if spec.strip() 

115 ] 

116 elif SCENARIOS_ENV_VAR in environ: 

117 configured = [ 

118 builder.source_for(spec=spec, base_dir=working_dir, origin=SCENARIOS_ENV_VAR) 

119 for spec in environ[SCENARIOS_ENV_VAR].split(os.pathsep) 

120 if spec.strip() 

121 ] 

122 else: 

123 configured = _pyproject_sources(project=project, builder=builder) 

124 return (*built_in_scenario_sources(), *configured)

127def path_source(path: Path, working_dir: Path, origin: str) -> ScenarioSource: 

128 """Return the file or directory source for ``path``, imported as a configured one is. 

129 

130 Raises :class:`ScenarioCatalogError` when ``path`` is neither an existing 

131 ``.py`` file nor a directory. 

132 """ 

133 project = find_project(working_dir=working_dir) 

134 builder = _SourceBuilder( 

135 project_root=project.root, 

136 skipped_paths=_skipped_paths( 

137 working_dir=working_dir, 

138 project_root=project.root, 

139 environ=os.environ, 

140 runs_dirs=(), 

141 ), 

142 ) 

143 resolved = (working_dir / path).resolve() 

144 if not resolved.exists(): 

145 raise ScenarioCatalogError(f"{path} does not exist.") 

146 return builder.source_for(spec=str(path), base_dir=working_dir, origin=origin)

176def child_process_environment() -> dict[str, str]: 

177 """Return this process's environment, set up for a ``glossogen`` child. 

178 

179 ``GLOSSOGEN_SCENARIOS`` lists the modules the current catalog loaded beyond 

180 the built-ins, by name, and ``PYTHONPATH`` starts with the directories the 

181 file and directory sources imported them from. The child imports the same 

182 modules under the same names, so a relative import inside a scenario package 

183 works there too, without resolving or scanning anything. 

184 """ 

185 catalog = current_scenario_catalog() 

186 environment = dict(os.environ) 

187 environment[SCENARIOS_ENV_VAR] = catalog.child_scenario_sources() 

188 python_path = catalog.child_import_roots() 

189 existing = environment.get(PYTHON_PATH_ENV_VAR) 

190 if existing: 

191 python_path.append(existing) 

192 if python_path: 

193 environment[PYTHON_PATH_ENV_VAR] = os.pathsep.join(python_path) 

194 return environment

197def _skipped_paths( 

198 working_dir: Path, 

199 project_root: Path, 

200 environ: Mapping[str, str], 

201 runs_dirs: Sequence[Path], 

202) -> frozenset[Path]: 

203 """Return the directories a directory source never descends into, resolved. 

204 

205 Each runs directory is read relative to the working directory and to the 

206 project root, so the project's ``runs/`` is skipped from any subdirectory. 

207 An absolute one is the same path either way. 

208 """ 

209 relative = [Path(DEFAULT_RUNS_DIR_NAME), *runs_dirs] 

210 configured_runs_dir = environ.get(RUNS_DIR_ENV_VAR) 

211 if configured_runs_dir: 

212 relative.append(Path(configured_runs_dir)) 

213 candidates = [Path(glossogen.__file__).parent] 

214 for runs_dir in relative: 

215 candidates.extend((working_dir / runs_dir, project_root / runs_dir)) 

216 return frozenset(candidate.resolve() for candidate in candidates)

274 def _import_root(self, path: Path) -> Path: 

275 """Return the directory ``path`` is imported relative to. 

276 

277 The project root when the path is inside it. Otherwise the path's parent, 

278 so a file outside the project imports as its own stem and a directory as 

279 its own name. 

280 """ 

281 if self._inside_project(path=path): 

282 return self.project_root 

283 return path.parent

285 def _inside_project(self, path: Path) -> bool: 

286 """Return whether ``path`` is the project root or under it.""" 

287 return path == self.project_root or self.project_root in path.parents
