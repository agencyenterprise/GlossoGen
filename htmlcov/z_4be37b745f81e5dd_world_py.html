1"""Round state for move_a_box: where each crate stands, its readings, and the verdicts. 

2 

3Every action goes into one ordered log per round. Each operator has a read 

4position in it, so the scenario can tell whether the other operator has acted 

5since this one last looked, which is what holds a lockstep action open. The log 

6itself is never rendered to an operator: the only trace of the other crate is

322 def resolve(self, round_number: int, trigger: str) -> RoundOutcome: 

323 """Close the round: the bay under each crate decides it, when the depot assigned it. 

324 

325 A bay takes one crate per floor, so when an operator leaves two crates on 

326 one bay only the lower slot scores its delivery. A crate slot converges 

327 when every operator's crate in that slot ended on one bay, and each 

328 converged slot pays every operator the bonus. 

329 """ 

330 state = self._round() 

331 final_bays: dict[str, int | None] = {} 

332 taken: dict[str, set[int]] = {agent_id: set() for agent_id in self._agent_ids} 

333 for crate_id in self._crate_ids: 

334 bay = self.scored_bay_of(crate_id=crate_id, cell=state.cells[crate_id]) 

335 owner = owner_of(crate_id) 

336 if bay is not None and bay in taken[owner]: 336 ↛ 337line 336 didn't jump to line 337 because the condition on line 336 was never true

337 bay = None 

338 if bay is not None: 

339 taken[owner].add(bay) 

340 final_bays[crate_id] = bay 

341 

342 matched_slots: list[int] = [] 

343 for slot in range(1, self._crates_each + 1): 

344 bays = [ 

345 final_bays[crate_id] for crate_id in self._crate_ids if slot_of(crate_id) == slot 

346 ] 

347 if None not in bays and len(set(bays)) == 1: 

348 matched_slots.append(slot) 

349 

350 points: dict[str, int] = {} 

351 for agent_id in self._agent_ids: 

352 scored = sum( 

353 DELIVERY_POINTS 

354 for crate_id in self._crates_of[agent_id] 

355 if final_bays[crate_id] is not None 

356 ) 

357 scored += MATCH_BONUS * len(matched_slots) 

358 points[agent_id] = scored 

359 

360 matched = len(matched_slots) == self._crates_each 

361 outcome = RoundOutcome( 

362 round_number=round_number, 

363 final_cells=dict(state.cells), 

364 final_bays=final_bays, 

365 temperatures={crate_id: readings[-1] for crate_id, readings in state.readings.items()}, 

366 steps_used=dict(state.steps_used), 

367 points=points, 

368 matched_slots=tuple(matched_slots), 

369 matched=matched, 

370 trigger=trigger, 

371 ) 

372 self._outcomes[round_number] = outcome 

373 self._open = None 

374 return outcome
