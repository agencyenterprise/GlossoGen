1"""The history an agent carries into a run from another finished run. 

2 

3A history-transfer run seats some of its agents with the history an agent built in 

4a source run, which may be of another scenario. That history is the source agent's 

5own: its system prompt, text, thinking, tool calls and tool results up to the cut. 

6One user message follows it and opens the new run with the new run's system and 

7initial prompts. The model keeps the source's system prompt, because pydantic-ai 

8adds an agent's system prompt only to an empty history. 

9 

10The history is built once, at launch, and saved in the run directory. A resumed run 

11and the protocol probes read it back and put it in front of what the agent did in 

12the run itself. The ATIF export and the thread export do not read it. 

13"""

37def build_source_history( 

38 source_events: list[SimulationEvent], 

39 source_agent_id: str, 

40 keep_thinking: bool, 

41 split_parallel_tool_calls: bool, 

42) -> list[ModelMessage]: 

43 """Rebuild the source agent's turns from ``source_events``, the source log up to the cut. 

44 

45 The history opens with the source agent's own system prompt and initial prompt. 

46 A tool call whose result was logged after the cut has no result in 

47 ``source_events``, so it is left out with its result. ``keep_thinking`` set to 

48 ``False`` leaves out the thinking parts, for a seat whose model did not write them. 

49 

50 Raises ``ValueError`` when the source agent took no turn before the cut. 

51 """ 

52 prompts = registered_prompts( 

53 registration=find_first_registration(events=source_events, agent_id=source_agent_id) 

54 ) 

55 history = build_message_history( 

56 events=source_events, 

57 agent_id=source_agent_id, 

58 system_prompt=prompts.full_system_prompt, 

59 runner_prompts=prompts.runner_prompts, 

60 target_timestamp=source_events[-1].timestamp, 

61 cutoff_round=None, 

62 tool_calls_only=False, 

63 channel_visibility={}, 

64 history_floor_round=None, 

65 filter_below_round=None, 

66 split_parallel_tool_calls=split_parallel_tool_calls, 

67 ) 

68 if not history: 

69 raise ValueError(f"source agent {source_agent_id!r} took no turn before the cut") 

70 if keep_thinking: 

71 return history 

72 return without_thinking(history=history)

75def without_thinking(history: list[ModelMessage]) -> list[ModelMessage]: 

76 """``history`` with every thinking part removed, and any response left empty dropped. 

77 

78 A provider takes back only the thinking that its own model wrote: an 

79 Anthropic block's signature or an OpenAI reasoning item. Thinking from another 

80 model reaches the seat as empty reasoning items. 

81 """ 

82 kept: list[ModelMessage] = [] 

83 for message in history: 

84 if not isinstance(message, ModelResponse): 

85 kept.append(message) 

86 continue 

87 parts = [part for part in message.parts if not isinstance(part, ThinkingPart)] 

88 if parts: 

89 kept.append(ModelResponse(parts=parts, timestamp=message.timestamp)) 

90 return kept

93def build_transferred_history( 

94 source_history: list[ModelMessage], 

95 opening_prompt: str, 

96) -> list[ModelMessage]: 

97 """The history a seat starts with: ``source_history``, then ``opening_prompt``. 

98 

99 The trailing ``ModelRequest`` holds ``opening_prompt``; the runner sends it as 

100 the agent's first request. 

101 """ 

102 return [*source_history, ModelRequest(parts=[UserPromptPart(content=opening_prompt)])]

105async def write_transferred_history(path: Path, history: list[ModelMessage]) -> None: 

106 """Save ``history`` to ``path`` as pydantic-ai's JSON message format.""" 

107 payload = ModelMessagesTypeAdapter.dump_json(history, indent=2) 

108 await asyncio.to_thread(path.write_bytes, payload)

111async def read_transferred_history(path: Path) -> list[ModelMessage]: 

112 """Load a history saved by :func:`write_transferred_history`.""" 

113 payload = await asyncio.to_thread(path.read_bytes) 

114 return ModelMessagesTypeAdapter.validate_json(payload)

117def append_own_turns( 

118 transferred: list[ModelMessage], own_history: list[ModelMessage] 

119) -> list[ModelMessage]: 

120 """The transferred history followed by the agent's turns in this run. 

121 

122 ``own_history`` is rebuilt from this run's log, so it opens with this run's 

123 system and initial prompts. The transferred history's last message replaced 

124 that opening for this agent, so it is dropped. 

125 """ 

126 return [*transferred, *own_history[1:]]

129def _seats_swapped_in_run(events: list[SimulationEvent]) -> frozenset[str]: 

130 """The seats that an in-run swap gave to another agent, in this run's log.""" 

131 return frozenset(event.agent_id for event in events if isinstance(event, AgentSwappedMidRun))

134def _transferred_seats( 

135 manifest: HistoryTransferManifest, events: list[SimulationEvent] 

136) -> dict[str, str]: 

137 """Each seat that still holds its transferred agent, mapped to its history file. 

138 

139 A seat that an in-run swap gave to another agent is left out: the agent swapped 

140 in did not start with the transferred history. 

141 """ 

142 swapped = _seats_swapped_in_run(events=events) 

143 return { 

144 seat.agent_id: seat.history_file for seat in manifest.seats if seat.agent_id not in swapped 

145 }

148async def splice_transferred_histories( 

149 run_dir: Path, events: list[SimulationEvent], state: RewindState 

150) -> RewindState: 

151 """Put each transferred seat's saved history in front of its rebuilt turns. 

152 

153 Returns ``state`` unchanged for a run without a history-transfer manifest. 

154 """ 

155 manifest = read_history_transfer_manifest(run_dir=run_dir) 

156 if manifest is None: 

157 return state 

158 histories = dict(state.agent_message_histories) 

159 for agent_id, history_file in _transferred_seats(manifest=manifest, events=events).items(): 

160 transferred = await read_transferred_history(path=run_dir / history_file) 

161 histories[agent_id] = append_own_turns( 

162 transferred=transferred, 

163 own_history=histories.get(agent_id, []), 

164 ) 

165 return state._replace(agent_message_histories=histories)

168async def with_transferred_history( 

169 run_dir: Path, 

170 events: list[SimulationEvent], 

171 agent_id: str, 

172 own_history: list[ModelMessage], 

173) -> list[ModelMessage]: 

174 """``own_history`` with the agent's saved transferred history in front of it. 

175 

176 ``own_history`` is rebuilt from the run's own log, as the protocol probes and 

177 the thread export do. It comes back unchanged for an agent that is not a 

178 transferred seat of the run in ``run_dir``. 

179 """ 

180 manifest = read_history_transfer_manifest(run_dir=run_dir) 

181 if manifest is None: 

182 return own_history 

183 history_file = _transferred_seats(manifest=manifest, events=events).get(agent_id) 

184 if history_file is None: 

185 return own_history 

186 transferred = await read_transferred_history(path=run_dir / history_file) 

187 return append_own_turns(transferred=transferred, own_history=own_history)
