1"""What a run needs in the environment before it can reach the models it calls. 

2 

3A run that cannot reach a model does not fail where it is launched. The 

4supervisor creates one task per agent runner and awaits them only after the game 

5clock has finished, so an agent that dies on its first call surfaces once the 

6whole configured duration has elapsed: `round_count` rounds of 

7`max_round_duration_seconds` each. The run directory is claimed and the JSONL 

8written either way, so what is left behind looks like a run that happened, and 

9the process exits 0. Checking the environment first turns that into an error at 

10the command line. 

11 

12Two things stop a caller getting to a model, and they behave identically. One is 

13a credential the environment does not carry. The other is `self-hosted` naming a 

14model that `SELF_HOSTED_BASE_URLS` does not serve, which is a spelling mistake 

15away at all times, since the map is keyed by the exact model string. 

16 

17Agents are not the only callers. A scenario that judges its own rounds calls a 

18model under a provider its own knobs name, `anthropic` for every judge shipped 

19here whatever the agents run under. That one hides better than the rest: the 

20judge is built on first use, so a run whose agents authenticate starts, spends, 

21and reaches its first judged action before anything goes wrong. 

22 

23A scheduled `swap_agent` hides better still. It names its own model and provider 

24and is built at a round boundary, so an unreachable one costs every round before 

25the swap, at full price, and then kills the agent it was meant to bring in. 

26 

27Where the run starts decides which of those boundaries it will ever cross. A 

28resumed run inherits its source's whole schedule and opens at `first_round`, and 

29the clock never visits what is below that, so demanding a credential for a swap 

30the run has already outlived refuses it for a model nothing will call. A swap 

31exactly at `first_round` does fire on resume, and is checked. 

32 

33The names below are the ones pydantic-ai accepts, which is narrower than what the 

34vendor SDKs read: `anthropic` builds a client from `ANTHROPIC_AUTH_TOKEN` alone, 

35and pydantic-ai refuses it, so listing that here would pass a run the platform 

36cannot start. A check that passes still leaves every failure those libraries can 

37report, such as a key that is set and rejected, or an endpoint that is listed and 

38down. 

39"""

88def require_reachable_models( 

89 scenario_cls: type[SimulationScenario], 

90 scenario_config: dict[str, Any], 

91 agent_overrides: dict[str, dict[str, str]] | None, 

92 default_model: str, 

93 default_provider: str, 

94 first_round: int, 

95) -> None: 

96 """Raise ValueError naming everything that would stop this run reaching a model. 

97 

98 Every flow that starts a run calls this before claiming a run directory, 

99 because a claimed directory is the thing the check exists to prevent. The 

100 flows differ in what they do with the error: the CLI exits on it, and the 

101 MCP ``start_run`` tool, which is the only caller of ``launch_simulation``, 

102 reports it as a tool error. No REST route starts a run. 

103 """ 

104 unreachable = find_unreachable_providers( 

105 consumers=resolve_agent_consumers( 

106 roles=scenario_cls.get_agent_roles(knobs=scenario_config), 

107 agent_overrides=agent_overrides, 

108 default_model=default_model, 

109 default_provider=default_provider, 

110 ) 

111 + scenario_cls.get_judge_models(knobs=scenario_config) 

112 + resolve_scheduled_swap_consumers( 

113 scenario_cls=scenario_cls, 

114 scenario_config=scenario_config, 

115 first_round=first_round, 

116 ) 

117 ) 

118 if unreachable: 

119 raise ValueError(describe_unreachable_providers(unreachable=unreachable))

122def resolve_agent_consumers( 

123 roles: list[AgentRole], 

124 agent_overrides: dict[str, dict[str, str]] | None, 

125 default_model: str, 

126 default_provider: str, 

127) -> tuple[ModelConsumer, ...]: 

128 """Return the model and provider each agent will run under, overrides applied. 

129 

130 Mirrors what the simulation does to the agents it builds, so the check 

131 covers what the run will actually call rather than what was named on the 

132 command line. ``agent_overrides`` is the normalized mapping, whose entries 

133 always carry both a model and a provider. 

134 """ 

135 if agent_overrides is None: 

136 overrides: dict[str, dict[str, str]] = {} 

137 else: 

138 overrides = agent_overrides 

139 resolved: list[ModelConsumer] = [] 

140 for role in roles: 

141 override = overrides.get(role.agent_id) 

142 if override is None: 

143 model = default_model 

144 provider = default_provider 

145 else: 

146 model = override["model"] 

147 provider = override["provider"] 

148 resolved.append(ModelConsumer(name=role.agent_id, model=model, provider=provider)) 

149 return tuple(resolved)

152def resolve_scheduled_swap_consumers( 

153 scenario_cls: type[SimulationScenario], 

154 scenario_config: dict[str, Any], 

155 first_round: int, 

156) -> tuple[ModelConsumer, ...]: 

157 """Return the model each scheduled swap brings in, from ``first_round`` on. 

158 

159 Read through the knobs model rather than off the raw config, so the entries 

160 are the same typed events the runtime dispatches rather than a second 

161 reading of their shape. 

162 

163 Boundaries below ``first_round`` are skipped because the clock will not 

164 visit them, which is the same rule the runtime applies to a schedule a 

165 resumed run inherited. ``>=`` rather than ``>``: a swap at the round a 

166 resume opens on does fire, since the cloned log stops before the source 

167 dispatched it. 

168 """ 

169 knobs = scenario_cls.knobs_model().model_validate(scenario_config) 

170 return tuple( 

171 ModelConsumer( 

172 name=f"{event.agent_id} swapped in at round {event.at_round}", 

173 model=event.model, 

174 provider=event.provider, 

175 ) 

176 for event in knobs.scheduled_events 

177 if isinstance(event, SwapAgent) and event.at_round >= first_round 

178 )

181def find_unreachable_providers( 

182 consumers: tuple[ModelConsumer, ...], 

183) -> tuple[UnreachableProvider, ...]: 

184 """Return every reason these callers would fail to reach a model.""" 

185 callers_by_problem: dict[tuple[str, str], list[str]] = {} 

186 for entry in consumers: 

187 for remedy in _remedies_for(model=entry.model, provider=entry.provider): 

188 callers_by_problem.setdefault((entry.provider, remedy), []).append(entry.name) 

189 return tuple( 

190 UnreachableProvider( 

191 provider=provider, 

192 caller_names=tuple(sorted(caller_names)), 

193 remedy=remedy, 

194 ) 

195 for (provider, remedy), caller_names in sorted(callers_by_problem.items()) 

196 )

199def describe_unreachable_providers(unreachable: tuple[UnreachableProvider, ...]) -> str: 

200 """Return the message naming what to do, per problem, and who it affects.""" 

201 lines = ["This run cannot reach every model it would call."] 

202 for entry in unreachable: 

203 callers = ", ".join(entry.caller_names) 

204 lines.append(f" {entry.provider} ({callers}): {entry.remedy}") 

205 lines.append("Commands read the nearest .env at or above the directory they run in.") 

206 return "\n".join(lines)

209def _remedies_for(model: str, provider: str) -> tuple[str, ...]: 

210 """Return what stands between this caller and its model, as things to go and do.""" 

211 if provider == SELF_HOSTED_PROVIDER: 

212 return _self_hosted_remedies(model=model) 

213 remedies: list[str] = [] 

214 for requirement in _REQUIREMENTS.get(provider, ()): 

215 if _any_name_carries_a_value(names=requirement.accepted_names): 

216 continue 

217 remedies.append(f"set {' or '.join(requirement.accepted_names)}") 

218 return tuple(remedies)

221def _self_hosted_remedies(model: str) -> tuple[str, ...]: 

222 """Return what a self-hosted caller is missing, endpoint map first, then key. 

223 

224 The map is read rather than only checked for presence, because a model it 

225 does not serve fails exactly like an unset variable, and is likelier: the 

226 keys are exact model strings, so serving one model and asking for another is 

227 a typo away. 

228 """ 

229 remedies: list[str] = [] 

230 raw = os.environ.get(SELF_HOSTED_BASE_URLS_VAR, "").strip() 

231 if raw == "": 

232 remedies.append(f"set {SELF_HOSTED_BASE_URLS_VAR}") 

233 else: 

234 remedies.extend(_endpoint_map_remedies(raw=raw, model=model)) 

235 if not _any_name_carries_a_value(names=(SELF_HOSTED_API_KEY_VAR,)): 

236 remedies.append(f"set {SELF_HOSTED_API_KEY_VAR}") 

237 return tuple(remedies)

240def _endpoint_map_remedies(raw: str, model: str) -> tuple[str, ...]: 

241 """Return what is wrong with the endpoint map for ``model``, if anything.""" 

242 served = _served_models(raw=raw) 

243 if served is None: 

244 return ( 

245 f"{SELF_HOSTED_BASE_URLS_VAR} is not a JSON object mapping model names " 

246 "to endpoint URLs", 

247 ) 

248 if not served: 

249 return (f"{SELF_HOSTED_BASE_URLS_VAR} lists no models",) 

250 if model not in served: 

251 return ( 

252 f"{SELF_HOSTED_BASE_URLS_VAR} has no endpoint for {model!r} " 

253 f"(it serves: {', '.join(served)})", 

254 ) 

255 return ()

258def _served_models(raw: str) -> tuple[str, ...] | None: 

259 """Return the model names the endpoint map declares, or None if it is not one.""" 

260 try: 

261 parsed = json.loads(raw) 

262 except json.JSONDecodeError: 

263 return None 

264 if not isinstance(parsed, dict): 264 ↛ 265line 264 didn't jump to line 265 because the condition on line 264 was never true

265 return None 

266 return tuple(sorted(str(name) for name in cast(dict[Any, Any], parsed)))

269def _any_name_carries_a_value(names: tuple[str, ...]) -> bool: 

270 """Return whether any of ``names`` is set to something other than blank. 

271 

272 A name that is set and empty counts as absent. `.env.example` ships every key 

273 with an empty value, so a copy of it that was never filled in is the likeliest 

274 way to arrive here, and the provider libraries reject it anyway. 

275 """ 

276 return any(os.environ.get(name, "").strip() != "" for name in names)

279def credential_variable_names() -> tuple[str, ...]: 

280 """Return every environment variable a run's reachability can depend on. 

281 

282 Derived from the table above rather than restated, so a provider added there is 

283 covered everywhere this list is used. `scenario_conformance` hides these while it 

284 builds a scenario, and a name missing from that guard is a check that passes on a 

285 machine holding the key and fails on one that does not. 

286 """ 

287 names = { 

288 name 

289 for requirements in _REQUIREMENTS.values() 

290 for requirement in requirements 

291 for name in requirement.accepted_names 

292 } 

293 return tuple(sorted(names | {SELF_HOSTED_BASE_URLS_VAR, SELF_HOSTED_API_KEY_VAR}))
